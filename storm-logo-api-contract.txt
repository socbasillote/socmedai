import type { RequestHandler } from "express";

import type { AuthenticatedUser } from "../services/userProfileService.js";

import {
  getWorkspaceLogoSettings,
  WorkspaceLogoServiceError,
} from "../services/workspaceLogoService.js";

import { uploadUserWorkspaceLogo } from "../services/workspaceLogoService.js";

import { WorkspaceLogoStorageError } from "../services/media/workspaceLogoStorage.js";

import { updateUserWorkspaceLogoSettings } from "../services/workspaceLogoService.js";

import { updateWorkspaceLogoSchema } from "../schemas/workspaceLogoUpdateSchema.js";

export const getWorkspaceLogo: RequestHandler = async (req, res, next) => {
  try {
    const workspaceId = req.params.workspaceId;

    if (typeof workspaceId !== "string") {
      res.status(404).json({
        error: "Workspace not found.",
      });
      return;
    }

    const authUser = res.locals.authUser as AuthenticatedUser;

    const logoOverlay = await getWorkspaceLogoSettings(authUser, workspaceId);

    res.status(200).json({
      logoOverlay,
    });
  } catch (error) {
    if (error instanceof WorkspaceLogoServiceError) {
      res.status(error.statusCode).json({
        error: error.message,
      });
      return;
    }

    next(error);
  }
};

export const uploadWorkspaceLogo: RequestHandler = async (req, res, next) => {
  try {
    const workspaceId = req.params.workspaceId;

    if (typeof workspaceId !== "string") {
      res.status(404).json({
        error: "Workspace not found.",
      });
      return;
    }

    // This route uses express.raw(), not JSON
    // or multipart/form-data.
    const contentType = req.headers["content-type"]
      ?.split(";")[0]
      ?.trim()
      .toLowerCase();

    const acceptedTypes = ["image/png", "image/jpeg", "image/webp"];

    if (!contentType || !acceptedTypes.includes(contentType)) {
      res.status(415).json({
        error: "Unsupported logo file type.",
      });
      return;
    }

    if (!Buffer.isBuffer(req.body)) {
      res.status(400).json({
        error: "Logo image bytes are required.",
      });
      return;
    }

    const logoOverlay = await uploadUserWorkspaceLogo(
      res.locals.authUser as AuthenticatedUser,
      workspaceId,
      req.body,
    );

    res.status(201).json({
      logoOverlay,
    });
  } catch (error) {
    if (error instanceof WorkspaceLogoServiceError) {
      res.status(error.statusCode).json({
        error: error.message,
      });
      return;
    }

    if (error instanceof WorkspaceLogoStorageError) {
      res.status(400).json({
        error: error.message,
      });
      return;
    }

    next(error);
  }
};

export const updateWorkspaceLogo: RequestHandler = async (req, res, next) => {
  try {
    const workspaceId = req.params.workspaceId;

    if (typeof workspaceId !== "string") {
      res.status(404).json({
        error: "Workspace not found.",
      });
      return;
    }

    const parsed = updateWorkspaceLogoSchema.safeParse(req.body);

    if (!parsed.success) {
      res.status(400).json({
        error: "Invalid logo settings.",
        message: parsed.error.issues[0]?.message ?? "Validation failed.",
      });
      return;
    }

    const logoOverlay = await updateUserWorkspaceLogoSettings(
      res.locals.authUser as AuthenticatedUser,
      workspaceId,
      parsed.data,
    );

    res.status(200).json({
      logoOverlay,
    });
  } catch (error) {
    if (error instanceof WorkspaceLogoServiceError) {
      res.status(error.statusCode).json({
        error: error.message,
      });
      return;
    }

    next(error);
  }
};

===== apps/api/src/services/workspaceLogoService.ts =====
import {
  getUserWorkspaceById,
  isValidWorkspaceId,
} from "./workspaceService.js";

import type { AuthenticatedUser } from "./userProfileService.js";

import {
  parseWorkspaceLogoSettings,
  type WorkspaceLogoSettings,
} from "./media/workspaceLogoConfig.js";

import Workspace from "../models/Workspace.js";

import { uploadWorkspaceLogo } from "./media/workspaceLogoStorage.js";

import {
  getPrivateImageBucket,
  type ImageStorageBucket,
} from "./media/supabaseImageStorage.js";

import type { UpdateWorkspaceLogoInput } from "../schemas/workspaceLogoUpdateSchema.js";

export class WorkspaceLogoServiceError extends Error {
  constructor(
    message: string,
    public readonly statusCode: number,
  ) {
    super(message);
    this.name = "WorkspaceLogoServiceError";
  }
}

export async function getWorkspaceLogoSettings(
  authUser: AuthenticatedUser,
  workspaceId: string,
): Promise<WorkspaceLogoSettings> {
  if (!isValidWorkspaceId(workspaceId)) {
    throw new WorkspaceLogoServiceError("Workspace not found.", 404);
  }

  // Existing service enforces ownerUserId
  // for the authenticated Supabase user.
  const workspace = await getUserWorkspaceById(authUser, workspaceId);

  if (!workspace) {
    throw new WorkspaceLogoServiceError("Workspace not found.", 404);
  }

  const settings =
    workspace.settings &&
    typeof workspace.settings === "object" &&
    !Array.isArray(workspace.settings)
      ? (workspace.settings as Record<string, unknown>)
      : {};

  return parseWorkspaceLogoSettings(settings.logoOverlay, workspaceId);
}

export async function uploadUserWorkspaceLogo(
  authUser: AuthenticatedUser,
  workspaceId: string,
  bytes: Uint8Array,
  bucket: ImageStorageBucket = getPrivateImageBucket(),
): Promise<WorkspaceLogoSettings> {
  if (!isValidWorkspaceId(workspaceId)) {
    throw new WorkspaceLogoServiceError("Workspace not found.", 404);
  }

  // Authorization must happen before any
  // upload to private object storage.
  const workspace = await getUserWorkspaceById(authUser, workspaceId);

  if (!workspace) {
    throw new WorkspaceLogoServiceError("Workspace not found.", 404);
  }

  if (
    !(bytes instanceof Uint8Array) ||
    bytes.byteLength === 0 ||
    bytes.byteLength > 5 * 1024 * 1024
  ) {
    throw new WorkspaceLogoServiceError("Invalid logo file size.", 400);
  }

  const existingSettings =
    workspace.settings &&
    typeof workspace.settings === "object" &&
    !Array.isArray(workspace.settings)
      ? (workspace.settings as Record<string, unknown>)
      : {};

  const currentLogo = parseWorkspaceLogoSettings(
    existingSettings.logoOverlay,
    workspaceId,
  );

  // The storage utility validates, normalizes,
  // uploads, and verifies the new private PNG.
  // Every upload creates an immutable version.
  const stored = await uploadWorkspaceLogo(
    {
      workspaceId,
      bytes,
    },
    bucket,
  );

  const updatedSettings = parseWorkspaceLogoSettings(
    {
      ...currentLogo,
      logo: {
        storageKey: stored.storageKey,
        sha256: stored.sha256,
      },
    },
    workspaceId,
  );

  // Recheck the original authorized owner.
  // Only change logoOverlay, not unrelated
  // workspace settings.
  const originalOverlay = existingSettings.logoOverlay;

  const guard =
    originalOverlay === undefined
      ? { "settings.logoOverlay": { $exists: false } }
      : { "settings.logoOverlay": originalOverlay };

  const updated = await Workspace.findOneAndUpdate(
    {
      _id: workspace._id,
      ownerUserId: workspace.ownerUserId,
      ...guard,
    },
    {
      $set: {
        "settings.logoOverlay": updatedSettings,
      },
    },
    {
      new: true,
      runValidators: true,
    },
  );

  if (!updated) {
    // The newly uploaded immutable object is
    // retained because storage has already
    // succeeded. Do not delete it blindly.
    throw new WorkspaceLogoServiceError(
      "Workspace logo settings changed during upload. " +
        "Refresh before uploading again.",
      409,
    );
  }

  return updatedSettings;
}

export async function updateUserWorkspaceLogoSettings(
  authUser: AuthenticatedUser,
  workspaceId: string,
  input: UpdateWorkspaceLogoInput,
): Promise<WorkspaceLogoSettings> {
  if (!isValidWorkspaceId(workspaceId)) {
    throw new WorkspaceLogoServiceError("Workspace not found.", 404);
  }

  const workspace = await getUserWorkspaceById(authUser, workspaceId);

  if (!workspace) {
    throw new WorkspaceLogoServiceError("Workspace not found.", 404);
  }

  const settings =
    workspace.settings &&
    typeof workspace.settings === "object" &&
    !Array.isArray(workspace.settings)
      ? (workspace.settings as Record<string, unknown>)
      : {};

  const previous = parseWorkspaceLogoSettings(
    settings.logoOverlay,
    workspaceId,
  );

  const nextSettings = parseWorkspaceLogoSettings(
    { ...previous, ...input },
    workspaceId,
  );

  // Optimistic concurrency guard.
  // The original configuration must still match
  // what we read before updating it.
  const originalOverlay = settings.logoOverlay;

  const guard =
    originalOverlay === undefined
      ? { "settings.logoOverlay": { $exists: false } }
      : { "settings.logoOverlay": originalOverlay };

  const updated = await Workspace.findOneAndUpdate(
    {
      _id: workspace._id,
      ownerUserId: workspace.ownerUserId,
      ...guard,
    },
    {
      $set: {
        "settings.logoOverlay": nextSettings,
      },
    },
    {
      new: true,
      runValidators: true,
    },
  );

  if (!updated) {
    throw new WorkspaceLogoServiceError(
      "Workspace logo settings changed. Refresh and retry.",
      409,
    );
  }

  return nextSettings;
}

import mongoose from "mongoose";
import { z } from "zod";

export const workspaceLogoPositionSchema = z.enum([
  "top-left",
  "top-right",
  "bottom-left",
  "bottom-right",
]);

const sha256Schema = z.string().regex(/^[a-f0-9]{64}$/);

const objectIdSchema = z.string().regex(/^[a-f0-9]{24}$/);

export const workspaceLogoSettingsSchema = z
  .object({
    enabled: z.boolean(),
    logo: z
      .object({
        storageKey: z.string().min(1).max(1024),
        sha256: sha256Schema,
      })
      .nullable(),
    position: workspaceLogoPositionSchema,
    sizePercent: z.number().int().min(5).max(30),
    opacity: z.number().min(0.1).max(1),
  })
  .strict();

export type WorkspaceLogoSettings = z.infer<typeof workspaceLogoSettingsSchema>;

export const DEFAULT_WORKSPACE_LOGO_SETTINGS: WorkspaceLogoSettings = {
  enabled: false,
  logo: null,
  position: "bottom-right",
  sizePercent: 20,
  opacity: 0.8,
};

export function getWorkspaceLogoStorageKey(
  workspaceId: string,
  logoId: string,
): string {
  const workspace = objectIdSchema.parse(workspaceId.toLowerCase());
  const logo = objectIdSchema.parse(logoId.toLowerCase());

  return `workspaces/${workspace}/logos/${logo}.png`;
}

export function parseWorkspaceLogoSettings(
  value: unknown,
  workspaceId: string,
): WorkspaceLogoSettings {
  if (!mongoose.isObjectIdOrHexString(workspaceId)) {
    throw new Error("Invalid workspace ID.");
  }

  const settings = workspaceLogoSettingsSchema.parse(
    value ?? DEFAULT_WORKSPACE_LOGO_SETTINGS,
  );

  if (settings.enabled && !settings.logo) {
    throw new Error("Enabled logo overlay requires an uploaded logo.");
  }

  if (settings.logo) {
    const prefix = `workspaces/${workspaceId.toLowerCase()}/logos/`;

    const expected = new RegExp(`^${prefix}[a-f0-9]{24}\\.png$`);

    if (!expected.test(settings.logo.storageKey)) {
      throw new Error("Logo does not belong to this workspace.");
    }
  }

  return settings;
}

/**
 * Return a deep value-copy suitable for persisting
 * on the admitted image operation.
 *
 * Call this at admission time, not at completion.
 */
export function snapshotWorkspaceLogoSettings(
  value: unknown,
  workspaceId: string,
): WorkspaceLogoSettings {
  return structuredClone(parseWorkspaceLogoSettings(value, workspaceId));
}

import { supabase } from "@/lib/supabase";

const API_BASE_URL =
  import.meta.env.VITE_API_URL?.replace(/\/+$/, "") || "http://localhost:4000";

export class ApiError extends Error {
  public readonly status: number;

  constructor(message: string, status: number) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

type ApiErrorResponse = {
  error?: string;
  message?: string;
};

export async function apiFetch<T>(
  path: string,
  init: RequestInit = {},
): Promise<T> {
  const { data, error: sessionError } = await supabase.auth.getSession();

  if (sessionError) {
    throw new ApiError("Unable to retrieve the authentication session.", 401);
  }

  const accessToken = data.session?.access_token;

  if (!accessToken) {
    throw new ApiError("You must sign in to access this resource.", 401);
  }

  const headers = new Headers(init.headers);
  headers.set("Authorization", `Bearer ${accessToken}`);

  if (init.body && !headers.has("Content-Type")) {
    headers.set("Content-Type", "application/json");
  }

  const normalizedPath = path.startsWith("/") ? path : `/${path}`;

  const response = await fetch(`${API_BASE_URL}${normalizedPath}`, {
    ...init,
    headers,
    credentials: init.credentials ?? "same-origin",
  });

  if (!response.ok) {
    let message = `API request failed with status ${response.status}.`;

    try {
      const body = (await response.json()) as ApiErrorResponse;
      message = body.message || body.error || message;
    } catch {
      // Keep the fallback message if the response isn't JSON.
    }

    throw new ApiError(message, response.status);
  }

  return (await response.json()) as T;
}

===== apps/web/src/features/image-studio/hooks/useWorkspaceLogoAvailability.ts =====
import { useEffect, useState } from "react";

import { getWorkspaces } from "@/lib/workspaces";

import { hasWorkspaceLogo } from "@/lib/workspaceLogoAvailability";

type LogoAvailability = "loading" | "available" | "unavailable" | "error";

export function useWorkspaceLogoAvailability(workspaceId: string | null) {
  const [result, setResult] = useState<{
    workspaceId: string | null;
    status: LogoAvailability;
  }>({
    workspaceId: null,
    status: "unavailable",
  });

  useEffect(() => {
    if (!workspaceId) {
      return;
    }

    let active = true;

    void getWorkspaces()
      .then((workspaces) => {
        if (!active) return;

        const selected = workspaces.find(
          (workspace) => workspace._id === workspaceId,
        );

        setResult({
          workspaceId,
          status:
            selected && hasWorkspaceLogo(selected)
              ? "available"
              : "unavailable",
        });
      })
      .catch(() => {
        if (!active) return;

        setResult({
          workspaceId,
          status: "error",
        });
      });

    return () => {
      active = false;
    };
  }, [workspaceId]);

  const status: LogoAvailability = !workspaceId
    ? "unavailable"
    : result.workspaceId === workspaceId
      ? result.status
      : "loading";

  return {
    status,
    logoAvailable: status === "available",
  };
}
