/**
 * Reuse the notice callback contract already
 * defined by the existing workspace actions hook.
 */
type NoticeHandler = Parameters<typeof useWorkspaceActions>[0]["setNotice"];

/**
 * Dashboard workspace orchestration.
 *
 * Responsibilities:
 * - Workspace loading and selection
 * - Workspace creation
 * - Create-workspace form state
 * - Workspace selector menu state
 * - Deriving the selected workspace
 *
 * Does not:
 * - Render any UI
 * - Manage Facebook or LinkedIn connections
 * - Handle OAuth callbacks
 * - Manage posts or publishing
 */
export function useDashboardWorkspace(setNotice: NoticeHandler) {
  // Existing workspace data and selection logic.
  const {
    workspaces,
    selectedWorkspaceId,
    setSelectedWorkspaceId,
    workspacesLoading,
    workspaceError,
    addWorkspace,
  } = useWorkspaces();

  // Existing workspace selector menu.
  const { workspaceMenuOpen, setWorkspaceMenuOpen } = useWorkspaceMenu();

  // Existing workspace creation form.
  const {
    createWorkspaceOpen,
    setCreateWorkspaceOpen,
    newWorkspaceName,
    setNewWorkspaceName,
    openCreateWorkspace,
    closeCreateWorkspace,
  } = useCreateWorkspaceForm();

  // Existing workspace creation action.
  const {
    creatingWorkspace,
    createWorkspaceError,
    setCreateWorkspaceError,
    handleCreateWorkspace,
  } = useWorkspaceActions({
    addWorkspace,
    setSelectedWorkspaceId,
    setNewWorkspaceName,
    setCreateWorkspaceOpen,
    setNotice,
  });

  // Derived workspace selection.
  const selectedWorkspace = useMemo(
    () =>
      workspaces.find((workspace) => workspace._id === selectedWorkspaceId) ??
      null,
    [workspaces, selectedWorkspaceId],
  );

  return {
    // Workspace data and selection.
    workspaces,
    selectedWorkspaceId,
    setSelectedWorkspaceId,
    workspacesLoading,
    workspaceError,
    selectedWorkspace,

    // Workspace selector menu.
    workspaceMenuOpen,
    setWorkspaceMenuOpen,

    // Workspace creation form.
    createWorkspaceOpen,
    setCreateWorkspaceOpen,
    newWorkspaceName,
    setNewWorkspaceName,
    openCreateWorkspace,
    closeCreateWorkspace,

    // Workspace creation action.
    creatingWorkspace,
    createWorkspaceError,
    setCreateWorkspaceError,
    handleCreateWorkspace,
  };
}

import { useState } from "react";
import { createWorkspace, type Workspace } from "@/lib/workspaces";
import { storeWorkspaceId } from "@/lib/workspaceSelection";

type UseWorkspaceActionsOptions = {
  addWorkspace: (workspace: Workspace) => void;
  setSelectedWorkspaceId: (workspaceId: string) => void;
  setNewWorkspaceName: (name: string) => void;
  setCreateWorkspaceOpen: (open: boolean) => void;
  setNotice: (message: string) => void;
};

export function useWorkspaceActions({
  addWorkspace,
  setSelectedWorkspaceId,
  setNewWorkspaceName,
  setCreateWorkspaceOpen,
  setNotice,
}: UseWorkspaceActionsOptions) {
  const [creatingWorkspace, setCreatingWorkspace] = useState(false);
  const [createWorkspaceError, setCreateWorkspaceError] = useState("");

  async function handleCreateWorkspace(newWorkspaceName: string) {
    const name = newWorkspaceName.trim();

    if (!name) {
      setCreateWorkspaceError("Workspace name is required.");
      return;
    }

    if (name.length > 100) {
      setCreateWorkspaceError(
        "Workspace name must be 100 characters or fewer.",
      );
      return;
    }

    setCreatingWorkspace(true);
    setCreateWorkspaceError("");

    try {
      const workspace = await createWorkspace(name);

      addWorkspace(workspace);
      setSelectedWorkspaceId(workspace._id);
      storeWorkspaceId(workspace._id);
      setNewWorkspaceName("");
      setCreateWorkspaceOpen(false);
      setNotice(`Workspace "${workspace.name}" created successfully.`);
    } catch (error) {
      setCreateWorkspaceError(
        error instanceof Error ? error.message : "Unable to create workspace.",
      );
    } finally {
      setCreatingWorkspace(false);
    }
  }

  return {
    creatingWorkspace,
    createWorkspaceError,
    setCreateWorkspaceError,
    handleCreateWorkspace,
  };
}

import { useEffect, useState } from "react";
import { getWorkspaces, type Workspace } from "@/lib/workspaces";
import {
  clearStoredWorkspaceId,
  readStoredWorkspaceId,
  storeWorkspaceId,
} from "@/lib/workspaceSelection";

export function useWorkspaces() {
  const [workspaces, setWorkspaces] = useState<Workspace[]>([]);
  const [selectedWorkspaceId, setSelectedWorkspaceId] = useState("");
  const [workspacesLoading, setWorkspacesLoading] = useState(true);
  const [workspaceError, setWorkspaceError] = useState("");

  useEffect(() => {
    let active = true;

    async function loadWorkspaces() {
      setWorkspacesLoading(true);
      setWorkspaceError("");

      try {
        const result = await getWorkspaces();

        if (!active) return;

        setWorkspaces(result);

        const storedId = readStoredWorkspaceId();

        const validStoredWorkspace = storedId
          ? result.find((workspace) => workspace._id === storedId)
          : undefined;

        const nextWorkspaceId =
          validStoredWorkspace?._id ?? result[0]?._id ?? "";

        setSelectedWorkspaceId(nextWorkspaceId);

        if (nextWorkspaceId) {
          storeWorkspaceId(nextWorkspaceId);
        } else {
          clearStoredWorkspaceId();
        }
      } catch (error) {
        if (!active) return;

        setWorkspaceError(
          error instanceof Error
            ? error.message
            : "Unable to load your workspaces.",
        );
      } finally {
        if (active) {
          setWorkspacesLoading(false);
        }
      }
    }

    void loadWorkspaces();

    return () => {
      active = false;
    };
  }, []);

  function addWorkspace(workspace: Workspace) {
    setWorkspaces((current) => [workspace, ...current]);
  }

  return {
    workspaces,
    selectedWorkspaceId,
    setSelectedWorkspaceId,
    workspacesLoading,
    workspaceError,
    addWorkspace,
  };
}

import {
  storeFacebookOAuthWorkspaceId,
  readFacebookOAuthWorkspaceId,
  clearFacebookOAuthWorkspaceId,

const WORKSPACE_A = "507f1f77bcf86cd799439011";
const WORKSPACE_B = "507f1f77bcf86cd799439012";

describe("facebookOAuthWorkspace", () => {
  beforeEach(() => {
    sessionStorage.clear();
  });

  it("stores the initiating Facebook OAuth workspace", () => {
    storeFacebookOAuthWorkspaceId(WORKSPACE_A);

    expect(sessionStorage.getItem("storm:facebook-oauth-workspace")).toBe(
      WORKSPACE_A,
    );
  });

  it("retrieves the original OAuth workspace", () => {
    storeFacebookOAuthWorkspaceId(WORKSPACE_A);

    const workspaceId = readFacebookOAuthWorkspaceId();

    expect(workspaceId).toBe(WORKSPACE_A);
  });

  it("clears the temporary OAuth workspace", () => {
    storeFacebookOAuthWorkspaceId(WORKSPACE_A);

    clearFacebookOAuthWorkspaceId();

    expect(readFacebookOAuthWorkspaceId()).toBeNull();

    expect(sessionStorage.getItem("storm:facebook-oauth-workspace")).toBeNull();
  });

  it("replaces the previous workspace when a new OAuth flow starts", () => {
    storeFacebookOAuthWorkspaceId(WORKSPACE_A);

    expect(readFacebookOAuthWorkspaceId()).toBe(WORKSPACE_A);

    storeFacebookOAuthWorkspaceId(WORKSPACE_B);

    expect(readFacebookOAuthWorkspaceId()).toBe(WORKSPACE_B);
  });

  it("rejects empty workspace IDs without replacing a valid workspace", () => {
    storeFacebookOAuthWorkspaceId(WORKSPACE_A);

    expect(() => {
      storeFacebookOAuthWorkspaceId("");
    }).toThrow("A workspace ID is required.");

    expect(() => {
      storeFacebookOAuthWorkspaceId("   ");
    }).toThrow("A workspace ID is required.");

    expect(readFacebookOAuthWorkspaceId()).toBe(WORKSPACE_A);
  });
});

const FACEBOOK_OAUTH_WORKSPACE_KEY = "storm:facebook-oauth-workspace";

/**
 * Remember which workspace initiated Facebook OAuth.
 *
 * This value is temporary and scoped to the browser tab.
 */
export function storeFacebookOAuthWorkspaceId(workspaceId: string): void {
  if (!workspaceId.trim()) {
    throw new Error("A workspace ID is required.");
  }

  try {
    sessionStorage.setItem(FACEBOOK_OAUTH_WORKSPACE_KEY, workspaceId);
  } catch {
    throw new Error(
      "Unable to save the Facebook OAuth workspace. Please enable browser session storage and try again.",
    );
  }
}

/**
 * Retrieve the workspace that initiated Facebook OAuth.
 */
export function readFacebookOAuthWorkspaceId(): string | null {
  try {
    return sessionStorage.getItem(FACEBOOK_OAUTH_WORKSPACE_KEY);
  } catch {
    return null;
  }
}

/**
 * Remove the temporary workspace reference.
 *
 * Call this after successful connection or when
 * the OAuth flow is explicitly cancelled.
 */
export function clearFacebookOAuthWorkspaceId(): void {
  try {
    sessionStorage.removeItem(FACEBOOK_OAUTH_WORKSPACE_KEY);
  } catch {
    // Ignore storage failures during cleanup.
  }
}

  workspace: Workspace;

    method: "POST",
    body: JSON.stringify({ name }),

  return response.workspace;

    return null;
  }

export function storeWorkspaceId(workspaceId: string): void {

    localStorage.setItem(STORAGE_KEY, workspaceId);
  } catch {
    // The app should still work if browser storage is unavailable.
  }
}

  } catch {
    // Ignore storage failures; selection can still work in memory.
