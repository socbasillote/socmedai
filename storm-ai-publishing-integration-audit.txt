import { useEffect, useRef, useState, type FormEvent } from "react";

import {
  createImageRequestKey,
  generateWorkspaceImage,
  getImageGenerationCredits,
  getWorkspaceImageStatus,
  getWorkspaceImageUrl,
  validateImagePrompt,
  validateImageBranding,
  type ImageBranding,
  type ImageBrandingMode,
  type ImageBrandingPosition,
  type GenerateImageCommand,
  type ImageAspectRatio,
  type ImageGenerationCredits,
  type ImageGenerationDetails,
} from "@/lib/imageGeneration";

import { ImageHistoryGallery } from "./ImageHistoryGallery";
import { BrandingPlacementPreview } from "./BrandingPlacementPreview";
import { WorkspaceLogoUploader } from "./WorkspaceLogoUploader";

import { useWorkspaceLogoAvailability } from "../hooks/useWorkspaceLogoAvailability";
import { snapshotImageGenerationCommand } from "@/lib/imageGenerationCommandSnapshot";

type ImageStudioProps = {
  selectedWorkspaceId: string | null;
};

const ASPECT_RATIOS: Array<{
  value: ImageAspectRatio;
  label: string;
  description: string;
}> = [
  {
    value: "square",
    label: "Square",
    description: "1:1",
  },
  {
    value: "portrait",
    label: "Portrait",
    description: "2:3",
  },
  {
    value: "landscape",
    label: "Landscape",
    description: "3:2",
  },
];

function errorMessage(error: unknown): string {
  return error instanceof Error
    ? error.message
    : "Something went wrong. Please try again.";
}

export function ImageStudio({ selectedWorkspaceId }: ImageStudioProps) {
  const [prompt, setPrompt] = useState("");
  const [aspectRatio, setAspectRatio] = useState<ImageAspectRatio>("square");

  const [brandingMode, setBrandingMode] = useState<ImageBrandingMode>("none");

  const [brandingText, setBrandingText] = useState("");

  const [brandingPosition, setBrandingPosition] =
    useState<ImageBrandingPosition>("bottom-right");

  const [brandingSize, setBrandingSize] = useState(20);

  const [brandingOpacity, setBrandingOpacity] = useState(80);

  const {
    status: logoAvailabilityStatus,
    logoAvailable,
    refresh: refreshLogoAvailability,
  } = useWorkspaceLogoAvailability(selectedWorkspaceId);

  const logoSelectionBlocked = brandingMode === "logo" && !logoAvailable;

  const [credits, setCredits] = useState<ImageGenerationCredits | null>(null);

  const [loadingCredits, setLoadingCredits] = useState(false);

  const [submitting, setSubmitting] = useState(false);

  const [details, setDetails] = useState<ImageGenerationDetails | null>(null);

  const [assetId, setAssetId] = useState<string | null>(null);

  const [imageUrl, setImageUrl] = useState<string | null>(null);

  const [error, setError] = useState("");
  const [creditsError, setCreditsError] = useState("");
  const [previewError, setPreviewError] = useState("");

  const [pendingRetry, setPendingRetry] = useState<GenerateImageCommand | null>(
    null,
  );

  // Invalidate results from requests associated
  // with an earlier workspace or generation.
  const generationRef = useRef(0);

  // Remount when the selected workspace changes
  // (the parent supplies a workspace-specific key).
  useEffect(() => {
    if (!selectedWorkspaceId) {
      setCredits(null);
      return;
    }

    let active = true;

    setLoadingCredits(true);
    setCreditsError("");

    void getImageGenerationCredits(selectedWorkspaceId)
      .then((value) => {
        if (active) setCredits(value);
      })
      .catch((cause: unknown) => {
        if (active) {
          setCredits(null);
          setCreditsError(errorMessage(cause));
        }
      })
      .finally(() => {
        if (active) setLoadingCredits(false);
      });

    return () => {
      active = false;
    };
  }, [selectedWorkspaceId]);

  async function refreshCredits() {
    if (!selectedWorkspaceId) return;

    try {
      const value = await getImageGenerationCredits(selectedWorkspaceId);

      setCredits(value);
      setCreditsError("");
    } catch (cause) {
      setCreditsError(errorMessage(cause));
    }
  }

  async function refreshStatus(id: string, version = generationRef.current) {
    if (!selectedWorkspaceId) return;

    const next = await getWorkspaceImageStatus(selectedWorkspaceId, id);

    if (version !== generationRef.current) return;

    setDetails(next);

    if (next.status === "ready") {
      const access = await getWorkspaceImageUrl(selectedWorkspaceId, id);

      if (version !== generationRef.current) return;

      setImageUrl(access.url);
      setPreviewError("");
    }

    if (next.status === "failed") {
      setImageUrl(null);
    }

    return next;
  }

  // Poll only an already-known pending operation.
  // Never repeat the generation POST while polling.
  useEffect(() => {
    if (!selectedWorkspaceId || !assetId || details?.state !== "processing") {
      return;
    }

    let active = true;
    let inFlight = false;

    const interval = window.setInterval(() => {
      if (!active || inFlight) return;

      inFlight = true;

      void refreshStatus(assetId)
        .catch((cause: unknown) => {
          if (active) {
            setError(
              `Unable to refresh generation status: ${errorMessage(cause)}`,
            );
          }
        })
        .finally(() => {
          inFlight = false;
        });
    }, 4000);

    return () => {
      active = false;
      window.clearInterval(interval);
    };
  }, [selectedWorkspaceId, assetId, details?.state]);

  async function submitCommand(command: GenerateImageCommand) {
    if (!selectedWorkspaceId || submitting) return;

    const version = ++generationRef.current;

    setSubmitting(true);
    setError("");

    try {
      const submitted = await generateWorkspaceImage(
        selectedWorkspaceId,
        command,
      );

      if (version !== generationRef.current) return;

      setAssetId(submitted.assetId);
      setPendingRetry(null);

      // Refresh even if the POST returned ready:
      // the GET endpoint supplies the safe
      // image metadata and readiness state.
      await refreshStatus(submitted.assetId, version);

      await refreshCredits();
    } catch (cause) {
      if (version !== generationRef.current) return;

      // Preserve the exact key and image settings.
      // A retry of an ambiguous POST must not
      // become a new credit-charged request.
      setPendingRetry(command);

      setError(errorMessage(cause));
      await refreshCredits();
    } finally {
      if (version === generationRef.current) {
        setSubmitting(false);
      }
    }
  }

  function buildSelectedBranding(): ImageBranding {
    if (brandingMode === "none") {
      return { type: "none" };
    }

    if (brandingMode === "logo" && !logoAvailable) {
      throw new Error("An enabled workspace logo is required.");
    }

    const appearance = {
      position: brandingPosition,
      sizePercent: brandingSize,
      opacity: brandingOpacity / 100,
    };

    if (brandingMode === "logo") {
      return validateImageBranding({
        type: "logo",
        ...appearance,
      });
    }

    return validateImageBranding({
      type: "text",
      text: brandingText,
      ...appearance,
    });
  }

  function handleGenerate(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (!selectedWorkspaceId || submitting) return;

    try {
      const normalizedPrompt = validateImagePrompt(prompt);

      const command: GenerateImageCommand = snapshotImageGenerationCommand({
        idempotencyKey: createImageRequestKey(),

        image: {
          prompt: normalizedPrompt,
          aspectRatio,
          branding: buildSelectedBranding(),
        },
      });

      // From this point, the command is immutable
      // for all retries of this logical request.
      setPendingRetry(null);
      setAssetId(null);
      setDetails(null);
      setImageUrl(null);
      setPreviewError("");
      setError("");

      void submitCommand(command);
    } catch (cause) {
      setError(errorMessage(cause));
    }
  }

  async function refreshPreview() {
    if (!selectedWorkspaceId || !assetId) return;

    try {
      const access = await getWorkspaceImageUrl(selectedWorkspaceId, assetId);

      setImageUrl(access.url);
      setPreviewError("");
    } catch (cause) {
      setPreviewError(errorMessage(cause));
    }
  }

  const availableCredits = credits?.credits.available;
  const costPerImage = credits?.credits.costPerImage;

  const insufficientCredits =
    availableCredits !== undefined &&
    costPerImage !== undefined &&
    availableCredits < costPerImage;

  const dailyLimitReached =
    credits !== null && credits.usage.attempts >= credits.usage.dailyLimit;

  const activeLimitReached =
    credits !== null && credits.usage.active >= credits.usage.maxActive;

  const cannotGenerate =
    !selectedWorkspaceId ||
    submitting ||
    pendingRetry !== null ||
    details?.state === "processing" ||
    details?.state === "action_required" ||
    !credits ||
    insufficientCredits ||
    dailyLimitReached ||
    activeLimitReached ||
    prompt.trim().length < 10 ||
    prompt.trim().length > 3000 ||
    (brandingMode === "text" &&
      (brandingText.trim().length === 0 ||
        brandingText.length > 80 ||
        /[\u0000-\u001f\u007f]/u.test(brandingText))) ||
    logoSelectionBlocked;

  return (
    <main className="mx-auto w-full max-w-6xl space-y-6 px-4 py-6 sm:px-6">
      <header className="space-y-2">
        <p className="text-xs font-semibold uppercase tracking-widest text-violet-600">
          AI Media
        </p>
        <h1 className="text-2xl font-bold text-slate-900">Image Studio</h1>
        <p className="text-sm text-slate-600">
          Create images using your Storm credits. Generated files are stored
          privately.
        </p>
      </header>

      <section className="grid gap-4 sm:grid-cols-3">
        <div className="rounded-2xl border border-slate-200 bg-white p-5">
          <p className="text-sm text-slate-500">Available credits</p>
          <p className="mt-2 text-3xl font-bold">
            {loadingCredits ? "..." : (availableCredits ?? "—")}
          </p>
          <button
            type="button"
            onClick={() => void refreshCredits()}
            disabled={!selectedWorkspaceId}
            className="mt-3 text-sm font-normal text-violet-700 hover:underline"
          >
            Refresh balance
          </button>
        </div>

        <div className="rounded-2xl border border-slate-200 bg-white p-5">
          <p className="text-sm text-slate-500">Cost per image</p>
          <p className="mt-2 text-3xl font-bold">{costPerImage ?? "—"}</p>
          <p className="mt-1 text-xs text-slate-500">Storm credits</p>
        </div>

        <div className="rounded-2xl border border-slate-200 bg-white p-5">
          <p className="text-sm text-slate-500">Today's usage (UTC)</p>
          <p className="mt-2 text-3xl font-bold">
            {credits
              ? `${credits.usage.attempts}/${credits.usage.dailyLimit}`
              : "—"}
          </p>
          <p className="mt-1 text-xs text-slate-500">
            Active operations:{" "}
            {credits
              ? `${credits.usage.active}/${credits.usage.maxActive}`
              : "—"}
          </p>
        </div>
      </section>

      {creditsError && (
        <p role="alert" className="text-sm text-red-700">
          Unable to load credits: {creditsError}
        </p>
      )}

      {!selectedWorkspaceId && (
        <p
          role="status"
          className="rounded-xl border border-amber-200 bg-amber-50 p-4 text-sm text-amber-800"
        >
          Select a workspace to use Image Studio.
        </p>
      )}

      <div className="grid gap-6 lg:grid-cols-2">
        <form
          onSubmit={handleGenerate}
          className="space-y-5 rounded-2xl border border-slate-200 bg-white p-5"
        >
          <div className="space-y-2">
            <label
              htmlFor="image-studio-prompt"
              className="text-sm font-semibold text-slate-800"
            >
              Image prompt
            </label>
            <textarea
              id="image-studio-prompt"
              rows={7}
              maxLength={3000}
              value={prompt}
              onChange={(event) => setPrompt(event.target.value)}
              placeholder="Describe the image, subject, lighting, composition, and style..."
              className="w-full resize-y rounded-xl border border-slate-200 p-3 text-sm outline-none focus:border-violet-500"
            />
            <p className="text-right text-xs text-slate-500">
              {prompt.length}/3000
            </p>
          </div>

          <fieldset className="space-y-3">
            <legend className="text-sm font-semibold text-slate-800">
              Aspect ratio
            </legend>

            <div className="grid grid-cols-3 gap-2">
              {ASPECT_RATIOS.map((item) => {
                const selected = aspectRatio === item.value;

                return (
                  <label
                    key={item.value}
                    className={`cursor-pointer rounded-xl border p-3 text-center ${
                      selected
                        ? "border-violet-500 bg-violet-50"
                        : "border-slate-200"
                    }`}
                  >
                    <input
                      type="radio"
                      name="aspectRatio"
                      value={item.value}
                      checked={selected}
                      onChange={() => setAspectRatio(item.value)}
                      className="sr-only"
                    />
                    <span className="block text-sm font-normal">
                      {item.label}

                    <span className="text-xs text-slate-500">
                      {item.description}

                  </label>
                );
              })}
            </div>
          </fieldset>

          <fieldset className="space-y-4 rounded-xl border border-slate-200 p-4">
            <legend className="px-2 text-sm font-semibold text-slate-800">
              Image branding (optional)

            <WorkspaceLogoUploader
              key={selectedWorkspaceId ?? "no-workspace"}
              workspaceId={selectedWorkspaceId}

                pendingRetry !== null ||
                details?.state === "processing" ||
                details?.state === "action_required"

              {(

                const disabled = item.value === "logo" && !logoAvailable;

                    key={item.value}

                      disabled
                        ? "cursor-not-allowed border-slate-200 bg-slate-100 text-slate-400"

                    <input

                onClick={() => void submitCommand(pendingRetry)}
                disabled={submitting}
                className="text-sm font-semibold text-amber-900 underline disabled:opacity-50"
              >
                Check / retry original request
              </button>
            </div>
          )}

          <p className="text-xs text-slate-500">
            The backend validates credits and limits again before admitting any
            generation.
          </p>
        </form>

        <section
          aria-label="Generated image preview"
          className="space-y-4 rounded-2xl border border-slate-200 bg-white p-5"
        >
          <div className="flex items-center justify-between gap-3">
            <h2 className="font-semibold text-slate-900">Preview</h2>
            <span className="rounded-full bg-slate-100 px-2 py-1 text-xs text-slate-700">
              {details?.state ?? "No image"}
            </span>
          </div>

          <div className="flex min-h-72 items-center justify-center overflow-hidden rounded-xl bg-slate-100">
            {imageUrl && details?.status === "ready" ? (
              <img
                src={imageUrl}
                alt="Generated image"
                referrerPolicy="no-referrer"
                onError={() =>
                  setPreviewError(
                    "The private image link expired or could not be loaded.",
                  )
                }
                className="max-h-[480px] w-full object-contain"
              />
            ) : details?.state === "processing" ? (
              <p
                role="status"
                className="p-5 text-center text-sm text-slate-600"
              >
                Processing image generation...
              </p>
            ) : details?.state === "action_required" ? (
              <p
                role="status"
                className="p-5 text-center text-sm text-amber-800"
              >
                This generation requires recovery. No additional generation will
                be started automatically.
              </p>
            ) : details?.status === "failed" ? (
              <p role="status" className="p-5 text-center text-sm text-red-700">
                Generation failed.
              </p>
            ) : (
              <p className="p-5 text-center text-sm text-slate-500">
                Your generated image will appear here.
              </p>
            )}
          </div>

          {previewError && (
            <p role="alert" className="text-sm text-red-700">
              {previewError}
            </p>
          )}

          {assetId && (
            <div className="flex flex-wrap gap-3">
              <button
                type="button"
                onClick={() =>
                  void refreshStatus(assetId).catch((cause: unknown) =>
                    setError(errorMessage(cause)),
                  )
                }
                className="rounded-xl border border-slate-200 px-4 py-2 text-sm font-normal"
              >
                Refresh status
              </button>

              {details?.status === "ready" && (
                <button
                  type="button"
                  onClick={() => void refreshPreview()}
                  className="rounded-xl border border-slate-200 px-4 py-2 text-sm font-normal"
                >
                  Refresh image link
                </button>
              )}
            </div>
          )}

          {details?.image && (
            <p className="text-xs text-slate-500">
              {details.image.width} × {details.image.height} · PNG
            </p>
          )}

          {details?.status === "ready" || details?.status === "failed" ? (
            <p className="text-xs text-slate-500">
              You can create another image using the prompt editor when your
              credit balance and workspace limits allow it.
            </p>
          ) : null}
        </section>
      </div>

      <ImageHistoryGallery
        key={selectedWorkspaceId ?? "no-workspace"}
        selectedWorkspaceId={selectedWorkspaceId}
        refreshKey={assetId}
      />
    </main>
  );
}

import { useEffect, useRef, useState } from "react";

import {
  getWorkspaceImageHistory,
  getWorkspaceImageStatus,
  getWorkspaceImageUrl,
  type ImageHistoryItem,
} from "@/lib/imageGeneration";

import {
  applyImageHistoryStatus,
  isImageStillProcessing,
} from "../lib/imageHistoryState";

type ImageHistoryGalleryProps = {
  selectedWorkspaceId: string | null;
  refreshKey?: string | null;
};

const PAGE_SIZE = 12;

const STATUS_POLL_INTERVAL_MS = 8000;

// Limit concurrent status requests.
// This protects the API when a workspace
// has multiple unfinished operations.
const MAX_STATUS_CHECKS_PER_CYCLE = 3;

function errorMessage(error: unknown): string {
  return error instanceof Error
    ? error.message
    : "Unable to complete this request.";
}

function imageStateLabel(item: ImageHistoryItem): string {
  switch (item.state) {
    case "ready":
      return "Ready";
    case "failed":
      return "Failed";
    case "action_required":
      return "Needs review";
    default:
      return "Processing";
  }
}

function mergeUnique(
  current: ImageHistoryItem[],
  incoming: ImageHistoryItem[],
): ImageHistoryItem[] {
  const ids = new Set<string>();

  return [...current, ...incoming].filter((item) => {
    if (ids.has(item.assetId)) return false;
    ids.add(item.assetId);
    return true;
  });
}

export function ImageHistoryGallery({
  selectedWorkspaceId,
  refreshKey,
}: ImageHistoryGalleryProps) {
  const [items, setItems] = useState<ImageHistoryItem[]>([]);

  const itemsRef = useRef<ImageHistoryItem[]>([]);

  // Always expose the latest rendered list to
  // the polling timer without restarting it
  // whenever a status changes.
  itemsRef.current = items;

  const [cursor, setCursor] = useState<string | null>(null);

  const [hasMore, setHasMore] = useState(false);

  const [loading, setLoading] = useState(false);

  const [loadingMore, setLoadingMore] = useState(false);

  const [error, setError] = useState("");

  const [selected, setSelected] = useState<ImageHistoryItem | null>(null);

  const [previewUrl, setPreviewUrl] = useState("");

  const [previewError, setPreviewError] = useState("");

  const [previewLoading, setPreviewLoading] = useState(false);

  // Incrementing a request version invalidates
  // older asynchronous results.
  const listVersion = useRef(0);
  const previewVersion = useRef(0);
  const currentWorkspace = useRef(selectedWorkspaceId);

  currentWorkspace.current = selectedWorkspaceId;

  const [manualRefresh, setManualRefresh] = useState(0);

  useEffect(() => {
    const version = ++listVersion.current;
    ++previewVersion.current;

    setItems([]);
    setCursor(null);
    setHasMore(false);
    setError("");
    setSelected(null);
    setPreviewUrl("");
    setPreviewError("");
    setPreviewLoading(false);
    setLoadingMore(false);

    if (!selectedWorkspaceId) {
      setLoading(false);

      return () => {
        ++listVersion.current;
        ++previewVersion.current;
      };
    }

    let active = true;

    setLoading(true);

    void getWorkspaceImageHistory(selectedWorkspaceId, { limit: PAGE_SIZE })
      .then((page) => {
        if (!active || version !== listVersion.current) {
          return;
        }

        setItems(page.images);
        setCursor(page.nextCursor);
        setHasMore(page.hasMore);
      })
      .catch((reason: unknown) => {
        if (!active || version !== listVersion.current) {
          return;
        }

        setError(errorMessage(reason));
      })
      .finally(() => {
        if (active && version === listVersion.current) {
          setLoading(false);
        }
      });

    return () => {
      active = false;
      ++listVersion.current;
      ++previewVersion.current;
    };
  }, [selectedWorkspaceId, refreshKey, manualRefresh]);

  useEffect(() => {
    if (!selectedWorkspaceId || loading) {
      return;
    }

    const workspace = selectedWorkspaceId;
    const version = listVersion.current;

    let active = true;
    let polling = false;
    let nextOffset = 0;

    async function pollPendingImages() {
      if (!active || polling) {
        return;
      }

      const pending = itemsRef.current.filter(isImageStillProcessing);

      if (pending.length === 0) {
        return;
      }

      polling = true;

      try {
        // Round-robin selection prevents older
        // pending jobs from starving when there
        // are more than 3 unfinished operations.
        const start = nextOffset % pending.length;

        const selectedJobs = Array.from(
          {
            length: Math.min(MAX_STATUS_CHECKS_PER_CYCLE, pending.length),
          },
          (_, index) => pending[(start + index) % pending.length]!,
        );

        nextOffset = (start + selectedJobs.length) % pending.length;

        // READ-ONLY GET requests.
        // Do not resubmit generation POST.
        const results = await Promise.allSettled(
          selectedJobs.map((item) =>
            getWorkspaceImageStatus(workspace, item.assetId),
          ),
        );

        if (
          !active ||
          version !== listVersion.current ||
          currentWorkspace.current !== workspace
        ) {
          return;
        }

        const updates = new Map(
          results.flatMap((result, index) => {
            if (result.status !== "fulfilled") {
              return [];
            }

            const requestedId = selectedJobs[index]?.assetId;

            // Never apply mismatched asset data.
            if (!requestedId || result.value.assetId !== requestedId) {
              return [];
            }

            return [[requestedId, result.value] as const];
          }),
        );

        if (updates.size === 0) {
          return;
        }

        setItems((previous) => {
          let changed = false;

          const updated = previous.map((item) => {
            const details = updates.get(item.assetId);

            if (!details) {
              return item;
            }

            const next = applyImageHistoryStatus(item, details);

            if (next !== item) {
              changed = true;
            }

            return next;
          });

          return changed ? updated : previous;
        });
      } finally {
        polling = false;
      }
    }

    // Check restored pending operations without
    // waiting for the first interval.
    void pollPendingImages();

    const interval = window.setInterval(() => {
      void pollPendingImages();
    }, STATUS_POLL_INTERVAL_MS);

    return () => {
      active = false;
      window.clearInterval(interval);
    };
  }, [selectedWorkspaceId, loading, refreshKey, manualRefresh]);

  function refreshHistory() {
    // Prevent a previous "load more" request
    // from modifying the new first page.
    ++listVersion.current;
    setManualRefresh((value) => value + 1);
  }

  async function loadMore() {
    if (!selectedWorkspaceId || loading || loadingMore || !hasMore || !cursor) {
      return;
    }

    const workspace = selectedWorkspaceId;
    const version = listVersion.current;

    setLoadingMore(true);
    setError("");

    try {
      const page = await getWorkspaceImageHistory(workspace, {
        limit: PAGE_SIZE,
        cursor,
      });

      if (
        version !== listVersion.current ||
        currentWorkspace.current !== workspace
      ) {
        return;
      }

      setItems((previous) => mergeUnique(previous, page.images));

      setCursor(page.nextCursor);
      setHasMore(page.hasMore);
    } catch (reason) {
      if (
        version === listVersion.current &&
        currentWorkspace.current === workspace
      ) {
        setError(errorMessage(reason));
      }
    } finally {
      if (
        version === listVersion.current &&
        currentWorkspace.current === workspace
      ) {
        setLoadingMore(false);
      }
    }
  }

  async function openImage(item: ImageHistoryItem) {
    if (!selectedWorkspaceId || item.state !== "ready") {
      return;
    }

    const workspace = selectedWorkspaceId;
    const version = ++previewVersion.current;

    setSelected(item);
    setPreviewUrl("");
    setPreviewError("");
    setPreviewLoading(true);

    try {
      const result = await getWorkspaceImageUrl(workspace, item.assetId);

      if (
        version !== previewVersion.current ||
        currentWorkspace.current !== workspace
      ) {
        return;
      }

      setPreviewUrl(result.url);
    } catch (reason) {
      if (
        version === previewVersion.current &&
        currentWorkspace.current === workspace
      ) {
        setPreviewError(errorMessage(reason));
      }
    } finally {
      if (
        version === previewVersion.current &&
        currentWorkspace.current === workspace
      ) {
        setPreviewLoading(false);
      }
    }
  }

  function closePreview() {
    ++previewVersion.current;
    setSelected(null);
    setPreviewUrl("");
    setPreviewError("");
    setPreviewLoading(false);
  }

  return (
    <section
      aria-label="Image history"
      className="space-y-4 rounded-2xl border border-slate-200 bg-white p-5"
    >
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div>
          <h2 className="text-lg font-semibold text-slate-900">
            Image history
          </h2>

          <p className="text-sm text-slate-500">
            Your previously generated workspace images.
          </p>
        </div>

        <button
          type="button"
          onClick={refreshHistory}
          disabled={!selectedWorkspaceId || loading}
          className="rounded-lg border border-slate-300 px-2 py-2 text-sm font-normal text-slate-700 hover:bg-slate-50 disabled:opacity-50"
        >
          Refresh history
        </button>
      </div>

      {!selectedWorkspaceId && (
        <p className="text-sm text-slate-500">
          Select a workspace to view its images.
        </p>
      )}

      {loading && (
        <p role="status" className="text-sm text-slate-500">
          Loading image history...
        </p>
      )}

      {error && (
        <p role="alert" className="text-sm text-red-700">
          {error}
        </p>
      )}

      {!loading && selectedWorkspaceId && items.length === 0 && !error && (
        <p className="rounded-xl bg-slate-50 p-5 text-sm text-slate-600">
          No image generations yet. Generate an image to see it here.
        </p>
      )}

      {items.length > 0 && (
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
          {items.map((item) => {
            const ready = item.state === "ready";

            return (
              <article
                key={item.assetId}
                className="overflow-hidden rounded-xl border border-slate-200"
              >
                <div className="flex aspect-square items-center justify-center bg-slate-100">
                  <div className="space-y-2 px-4 text-center">
                    <p className="text-sm font-semibold text-slate-700">
                      {imageStateLabel(item)}
                    </p>

                    <p className="text-xs text-slate-500">{item.aspectRatio}</p>

                    {ready && (
                      <button
                        type="button"
                        onClick={() => {
                          void openImage(item);
                        }}
                        className="rounded-lg bg-violet-600 px-2 py-2 text-sm font-normal text-white hover:bg-violet-700"
                        aria-label={`Open image ${item.assetId}`}
                      >
                        Open image
                      </button>
                    )}
                  </div>
                </div>

                <div className="space-y-1 p-3">
                  <p className="text-xs text-slate-500">

                  </p>

            type="button"
            disabled={loading || loadingMore || !selectedWorkspaceId}
            onClick={() => {

            }}
            className="rounded-lg border border-slate-300 px-4 py-2 text-sm font-normal text-slate-700 hover:bg-slate-50 disabled:opacity-50"
          >
            {loadingMore ? "Loading..." : "Load more"}
          </button>
        </div>
      )}

      {selected && (

          aria-label="Generated image preview"

            <div className="flex items-center justify-between gap-3">
              <h3 className="font-semibold text-slate-900">Generated image</h3>

              <button
                type="button"
                onClick={closePreview}
                className="rounded-lg border px-2 py-2 text-sm"
              >
                Close
              </button>
            </div>

            {previewLoading && <p role="status">Loading private preview...</p>}

            {previewError && (
              <p role="alert" className="text-sm text-red-700">
                {previewError}
              </p>
            )}

            {previewUrl && (
              <img
                src={previewUrl}
                alt="Previously generated image"
                referrerPolicy="no-referrer"
                className="mx-auto max-h-[65vh] max-w-full rounded-lg object-contain"

                  setPreviewError(
                    "The private preview could not be loaded. Refresh its access URL.",

                }}
              />
            )}

            <div className="flex flex-wrap gap-2">
              <button
                type="button"
                disabled={previewLoading}

            </div>

            <p className="text-xs text-slate-500">
              Private preview URLs expire. Refresh access if the image stops
              loading.
            </p>
          </div>
        </div>
      )}
    </section>
  );
}

      {error && (
        <p role="alert" className="text-sm text-rose-600">
          {error}
        </p>
      )}
    </form>
  );

      !connectionId

      setError(
        "Publishing requirements changed. Review your workspace, " +
          "caption, selected platform, and publishing connection.",
      );
      return;

    const platformLabel =
      publishingPlatform === "facebook" ? "Facebook" : "LinkedIn";

      [publishingPlatform]: connectionId,

          title: title.trim(),
          caption: caption.trim(),
          ...getPostMediaUpdate(mediaState),

          workspaceId: selectedWorkspace._id,

      // The user may have changed workspaces or posts while

          `Check your ${platformLabel} account and Posts page before taking further action.`,
      );
      setShowPublishConfirmation(false);

        err instanceof Error ? err.message : "An unexpected error occurred.";

      if (publishRequestStarted) {
        setPublishingOutcome("unknown");
        setPublishingMessage(
          `Storm could not confirm the ${platformLabel} publishing outcome. ` +
            `Check your ${platformLabel} account and Posts page before retrying. ` +
            errorMessage,
        );
      } else if (savedPostId) {
        setPublishingOutcome("failed");
        setPublishingMessage(
          "The publishing request was not started. " +
            "Your saved draft remains in Storm. " +
            errorMessage,
        );
      } else {
        setPublishingOutcome("failed");
        setPublishingMessage(
          mode === "edit"
            ? "Storm could not confirm that your draft changes were saved. " +
                "No publishing request was started. Check Posts before retrying. " +
                errorMessage
            : "Storm could not confirm that your draft was created. " +
                "No publishing request was started. Check Posts before retrying. " +
                errorMessage,
        );
      }

      setShowPublishConfirmation(false);
    } finally {
      setPublishing(false);
      setSaving(false);

      if (!publishRequestStarted && !savedPostId) {
        publishingLockRef.current = false;
      }
    }
  }

  function handlePublishReview() {
    if (
      saving ||
      publishing ||
      publishingLockRef.current ||
      savedPublishingPostId ||
      showPublishConfirmation
    ) {
      if (savedPublishingPostId) {
        setError(
          "This post already has a saved publishing attempt. " +
            "Check your Posts page before taking further action.",
        );
      }
      return;
    }

    setError("");

    if (!workspaceContext.valid) {
      setError(
        workspaceContext.message ?? "Unable to verify the post workspace.",
      );
      return;
    }

    if (selectedMediaAssetId) {
      setError(
        "Media publishing is not supported yet. " +
          "Remove the selected image before publishing.",
      );
      return;
    }

    if (mode === "edit" && !postId) {
      setError("Post ID is missing.");
      return;
    }

    if (workspacesLoading) {
      setError("Please wait while your workspace loads.");
      return;
    }

    if (workspaceError || !selectedWorkspace) {
      setError("Select a valid workspace before publishing.");
      return;
    }

    if (!caption.trim()) {
      setError("Caption is required.");
      return;
    }

    if (selectedPlatforms.length !== 1) {
      setError("Select exactly one platform for Publish Now.");
      return;
    }

    const platform = selectedPlatforms[0];

    if (platform === "linkedin") {
      if (linkedInStatus !== "connected" || !linkedInConnectionId) {
        setError("Connect a LinkedIn account before publishing.");
        return;
      }
    } else if (platform === "facebook") {
      const validSelection = facebookConnections.some(
        (connection) =>
          connection.id === selectedFacebookConnectionId &&
          connection.status === "connected" &&
          connection.canPublish,
      );

      if (
        facebookConnectionsLoading ||
        facebookConnectionsError ||
        !validSelection
      ) {
        setError(
          "Select a connected Facebook Page with publishing permission.",
        );
        return;
      }
    } else {
      setError("Publish Now does not support this platform yet.");
      return;
    }

    setShowPublishConfirmation(true);
  }

  function handleCancel() {
    if (publishing || saving) {
      return;
    }

    navigate(plannerReturnUrl ?? "/posts");
  }

  const publishingPlatformLabel =
    selectedPlatforms[0] === "facebook" ? "Facebook" : "LinkedIn";

  // Phase 13.3.1:
  // Derive checklist data from the existing editor state.
  // Do not replace the current publishing submission flow.

  const timezone = Intl.DateTimeFormat().resolvedOptions().timeZone || "UTC";

  let checklistDraft = createPublishingDraft(
    selectedWorkspaceId ?? "",
    timezone,
  );

  checklistDraft = setSharedCaption(checklistDraft, caption);

  // Readiness must reflect the CURRENT editor selection,
  // including media added to a new draft, replacement,
  // or explicit removal from an existing draft.
  checklistDraft = setSharedMedia(
    checklistDraft,
    mediaState.selectedMediaIds.map((assetId) => ({
      assetId,
      type: "image" as const,
    })),
  );

  const accountCapabilities: PublishingCapabilityContext = {
    accounts: {},
  };

  for (const platform of selectedPlatforms) {
    let connectionId = "";
    let accountName: string = platform;
    let format: PublishingDestination["format"] = "text";
    let connected = false;
    let canPublish = false;

    if (platform === "facebook") {
      const connection = facebookConnections.find(
        (item) => item.id === selectedFacebookConnectionId,
      );

      connectionId = connection?.id ?? "";
      accountName = connection?.pageName ?? "Select a Facebook Page";

      connected = connection?.status === "connected";
      canPublish = connection?.canPublish === true;
    }

    if (platform === "linkedin") {
      connectionId = linkedInConnectionId ?? "";
      accountName = linkedInDisplayName ?? "LinkedIn account";

      connected =
        linkedInStatus === "connected" && Boolean(linkedInConnectionId);

      canPublish = connected;
    }

    if (platform === "instagram") {
      accountName = "Instagram connection unavailable";
      format = "image";
    }

    const destination: PublishingDestination = {
      id: `${platform}:${connectionId || "unconnected"}`,
      platform,
      connectionId,
      accountName,
      format,
    };

    checklistDraft = addDestination(checklistDraft, destination);

    // Instagram is intentionally not given a verified
    // publishing account until its integration exists.
    if (platform !== "instagram" && connectionId) {
      accountCapabilities.accounts[destination.id] = {
        connectionId,
        connected,
        canPublish,
      };
    }
  }

  if (postMode === "scheduled" && scheduledAt) {
    const parsed = parseLocalScheduleDate(scheduledAt);

    checklistDraft.publishing.scheduledAt = parsed
      ? toScheduledAtISOString(parsed)
      : "";
  }

  const readiness = evaluatePublishingReadiness(
    checklistDraft,
    accountCapabilities,
    postMode === "immediate"
      ? "now"
      : postMode === "scheduled"
        ? "schedule"
        : "draft",
  );

  const checklistBlockers: string[] = [];

  if (!workspaceContext.valid) {
    checklistBlockers.push(
      workspaceContext.message ?? "Unable to verify the post workspace.",
    );
  }

  if (mode === "edit" && postMode !== "draft") {
    checklistBlockers.push(
      "Scheduling an existing draft is not supported by " +
        "the current editor.",
    );
  }

  if (
    postMode === "immediate" &&
    (workspacesLoading || workspaceError || !selectedWorkspace)
  ) {
    checklistBlockers.push(
      "Select a verified active workspace before publishing.",
    );
  }

  const editorContent = loadingPost ? (
    <div
      role="status"
      className="rounded-xl border border-slate-200 bg-slate-50 p-8 text-center"
    >
      <p className="text-sm text-slate-600">Loading post...</p>
    </div>
  ) : loadError ? (
    <div
      role="alert"
      className="rounded-xl border border-rose-200 bg-rose-50 p-6"
    >
      <h2 className="font-semibold text-rose-800">Unable to edit post</h2>

      <p className="mt-2 text-sm text-rose-700">{loadError}</p>

      <button
        type="button"
        onClick={handleCancel}
        className="mt-4 rounded-lg bg-white px-4 py-2 text-sm font-normal text-slate-700 shadow-sm"
      >
        Back to Posts
      </button>
    </div>
  ) : (
    <PostEditorForm
      mode={mode}
      locked={savedPublishingPostId !== null}
      title={title}
      caption={caption}
      selectedPlatforms={selectedPlatforms}
      postMode={postMode}
      scheduledAt={scheduledAt}
      minScheduledAt={minScheduledAt}
      saving={saving || publishing}
      error={error}
      linkedInConnectionId={
        linkedInStatus === "connected" ? linkedInConnectionId || null : null
      }
      linkedInDisplayName={linkedInDisplayName || null}
      linkedInStatus={linkedInStatus}
      onTitleChange={(value) => {
        setTitle(value);
        setError("");
      }}
      onCaptionChange={(value) => {
        setCaption(value);
        setError("");
      }}
      onPlatformsChange={(platforms) => {
        setSelectedPlatforms(platforms);
        setError("");
      }}
      onPostModeChange={(value) => {
        setPostMode(value);
        setError("");
      }}
      onScheduledAtChange={(value) => {
        setScheduledAt(value);
        setError("");
      }}
      facebookConnections={facebookConnections}
      selectedFacebookConnectionId={
        facebookConnections.some(
          (connection) =>
            connection.id === selectedFacebookConnectionId &&
            connection.canPublish &&
            connection.status === "connected",
        )
          ? selectedFacebookConnectionId
          : null
      }
      facebookConnectionsLoading={facebookConnectionsLoading}
      facebookConnectionsError={facebookConnectionsError}
      onFacebookConnectionChange={(connectionId) => {
        setSelectedFacebookConnectionId(connectionId);
        setError("");
      }}
      workspaceId={
        workspaceContext.valid
          ? mode === "edit"
            ? postWorkspaceId
            : selectedWorkspaceId
          : null
      }
      selectedMediaAssetId={selectedMediaAssetId}
      onMediaSelect={handleMediaSelect}
      onMediaRemove={handleMediaRemove}
      onSave={handleSave}

  );

  return (

                    setError("");
                  }}
                  onScheduledAtChange={(value) => {
                    setScheduledAt(value);
                    setError("");
                  }}
                  onCancel={handleCancel}
                />
              </div>
            ) : undefined
          }
          preview={
            !loadingPost && !loadError ? (
              <PostPreview
                title={title}
                caption={caption}
                platforms={selectedPlatforms}
              />
            ) : undefined
          }
        >
          <div className="mb-6 rounded-xl border border-slate-200 bg-slate-50 p-4">
            <p className="text-xs font-normal uppercase tracking-wide text-slate-500">
              Active Workspace
            </p>

            {workspacesLoading ? (
              <p className="mt-2 text-sm text-slate-600">
                Loading workspace...
              </p>

              <p role="alert" className="mt-2 text-sm text-rose-600">

              role="alert"

              <p className="text-sm text-rose-700">

          )}

          {publishing && (

              role="status"

            >
              <div className="flex items-center gap-3">
                <div
                  aria-hidden="true"
                  className="h-5 w-5 animate-spin rounded-full border-2 border-violet-200 border-t-violet-600"

          className="fixed inset-0 z-50 flex items-center justify-center bg-slate-950/60 p-4"
          role="presentation"
        >
          <div
            role="dialog"
            aria-modal="true"
            aria-labelledby="publish-confirmation-title"
            aria-describedby="publish-confirmation-description"
            className="w-full max-w-md rounded-2xl bg-white p-6 shadow-2xl"

            <h2
              id="publish-confirmation-title"
              className="text-xl font-bold text-slate-900"
            >
              Publish to {publishingPlatformLabel}?
            </h2>

            <p
              id="publish-confirmation-description"
              className="mt-3 text-sm leading-6 text-slate-600"
            >
              This will immediately publish your post to your selected{" "}
              {publishingPlatformLabel} destination.
            </p>

            <div className="mt-5 rounded-xl border border-slate-200 bg-slate-50 p-4">
              <p className="text-xs font-semibold uppercase tracking-wide text-slate-500">
                Publishing Account
              </p>

              <p className="mt-2 text-sm font-semibold text-slate-900">
                {linkedInDisplayName || "Connected LinkedIn account"}
              </p>

              <p className="mt-4 text-xs font-semibold uppercase tracking-wide text-slate-500">
                Post Preview
              </p>

              <p className="mt-2 max-h-40 overflow-y-auto whitespace-pre-wrap break-words text-sm text-slate-700">
                {caption.trim()}
              </p>
            </div>

            <p className="mt-4 text-xs leading-5 text-amber-700">
              Once published, this post may be visible to other{" "}
              {publishingPlatformLabel} users. Publishing cannot be undone from
              Storm.
            </p>

            <div className="mt-6 flex justify-end gap-3">
              <button
                type="button"
                onClick={() => setShowPublishConfirmation(false)}
                disabled={publishing || saving}
                className="rounded-xl border border-slate-200 px-4 py-2.5 text-sm font-semibold text-slate-700 transition hover:bg-slate-50 disabled:cursor-not-allowed disabled:opacity-50"
              >
                Cancel
              </button>

              <button
                type="button"
                onClick={() => {
                  void handleConfirmPublish();
                }}
                disabled={publishing || saving}
                className="rounded-xl bg-violet-600 px-4 py-2.5 text-sm font-semibold text-white transition hover:bg-violet-700 disabled:cursor-not-allowed disabled:opacity-50"
              >
                {publishing ? "Publishing..." : "Confirm Publish"}
              </button>
            </div>
          </div>
        </div>
      )}
    </main>
  );
}

import { useEffect, useState } from "react";
import { CheckCircle2, ImageIcon, RefreshCw } from "lucide-react";

import { getWorkspaceImageUrl } from "@/lib/imageGeneration";

import {
  getMediaLibrary,
  type MediaLibraryItem,
  type MediaLibraryScope,
} from "@/lib/mediaLibrary";

type LibraryImage = MediaLibraryItem & {
  previewUrl: string | null;
};

type Props = {
  workspaceId: string | null;
  selectedAssetId: string | null;
  disabled?: boolean;
  onSelect: (assetId: string) => void;
};

export function MediaLibraryPicker({
  workspaceId,
  selectedAssetId,
  disabled = false,
  onSelect,
}: Props) {
  const [scope, setScope] = useState<MediaLibraryScope>("current");

  const [images, setImages] = useState<LibraryImage[]>([]);

  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [refreshVersion, setRefreshVersion] = useState(0);

  useEffect(() => {
    // A new active workspace should always
    // start with the Current Workspace filter.
    setScope("current");
    setImages([]);
    setError("");
  }, [workspaceId]);

  useEffect(() => {
    let active = true;

    setImages([]);
    setError("");

    if (!workspaceId) {
      setLoading(false);

      return () => {
        active = false;
      };
    }

    setLoading(true);

    async function loadImages() {
      if (!workspaceId) return;

      try {
        const response = await getMediaLibrary(workspaceId, scope, 12);

        if (!active) return;

        // Every asset carries its source workspace.
        // Signed URLs must be requested against
        // THAT workspace, not the active workspace.
        const results = await Promise.all(
          response.images.map(async (image) => {
            try {
              const access = await getWorkspaceImageUrl(
                image.sourceWorkspaceId,
                image.assetId,
              );

              return {
                ...image,
                previewUrl: access.url,
              };
            } catch {
              return {
                ...image,
                previewUrl: null,
              };
            }
          }),
        );

        if (!active) return;

        setImages(results);
      } catch {
        if (!active) return;

        setError("Unable to load Media Library images. Try again.");
      } finally {
        if (active) {
          setLoading(false);
        }
      }
    }

    void loadImages();

    return () => {
      active = false;
    };
  }, [workspaceId, scope, refreshVersion]);

  function changeScope(nextScope: MediaLibraryScope) {
    if (nextScope === scope) return;

    setScope(nextScope);
  }

  return (
    <section aria-label="Media Library picker" className="space-y-4">
      <div className="flex items-center justify-between gap-3">
        <div>
          <h3 className="text-sm font-semibold text-slate-900">
            Media Library
          </h3>

          <p className="mt-1 text-xs text-slate-500">
            Reuse images without generating them again.
          </p>
        </div>

        <button
          type="button"
          onClick={() => setRefreshVersion((value) => value + 1)}
          disabled={loading || disabled || !workspaceId}
          className="inline-flex items-center gap-1 rounded-lg border border-slate-200 px-2 py-2 text-xs font-normal text-slate-700 disabled:opacity-50"
        >
          <RefreshCw size={14} aria-hidden="true" />
          Refresh
        </button>
      </div>

      <fieldset disabled={disabled || !workspaceId} className="space-y-2">
        <legend className="text-xs font-semibold text-slate-700">
          Show images from
        </legend>

        <div className="grid grid-cols-2 gap-2">
          {(
            [
              {
                value: "current",
                label: "Current Workspace",
              },
              {
                value: "all",
                label: "All Workspaces",
              },
            ] as const
          ).map((option) => (
            <button
              key={option.value}
              type="button"
              aria-pressed={scope === option.value}
              onClick={() => changeScope(option.value)}
              className={[
                "rounded-xl border px-2 py-2.5",
                "text-xs font-semibold transition",
                "disabled:opacity-50",
                scope === option.value
                  ? "border-violet-300 bg-violet-50 text-violet-700"
                  : "border-slate-200 bg-white text-slate-600 hover:bg-slate-50",
              ].join(" ")}
            >
              {option.label}
            </button>
          ))}
        </div>
      </fieldset>

      <p className="text-xs text-slate-500">
        {scope === "current"
          ? "Showing images from your active workspace."
          : "Showing images from all workspaces you own."}
      </p>

      {!workspaceId && (
        <p className="rounded-lg bg-slate-50 p-4 text-sm text-slate-600">
          Select a workspace to view its images.
        </p>
      )}

      {loading && (
        <p role="status" className="text-sm text-slate-500">
          Loading images...
        </p>
      )}

      {error && (
        <p role="alert" className="text-sm text-rose-700">
          {error}
        </p>
      )}

      {!loading && !error && workspaceId && images.length === 0 && (
        <div className="rounded-xl border border-dashed border-slate-200 p-6 text-center">
          <ImageIcon size={28} className="mx-auto text-slate-400" />

          <p className="mt-2 text-sm font-normal text-slate-700">
            No ready images found
          </p>

          <p className="mt-1 text-xs text-slate-500">
            {scope === "current"
              ? "Try All Workspaces or generate a new image."
              : "Generate an image in Image Studio, then refresh."}
          </p>
        </div>
      )}

      {!loading && images.length > 0 && (
        <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
          {images.map((image) => {
            const selected = selectedAssetId === image.assetId;

            return (
              <div key={image.assetId} className="min-w-0 space-y-2">
                <button
                  type="button"
                  aria-label={`Select image ${image.assetId}`}
                  aria-pressed={selected}
                  disabled={disabled || !image.previewUrl || !workspaceId}
                  onClick={() => onSelect(image.assetId)}
                  className={[
                    "relative block aspect-square w-full",
                    "overflow-hidden rounded-xl border-2",
                    "bg-slate-50 transition",
                    "focus-visible:outline",
                    "focus-visible:outline-2",
                    "focus-visible:outline-violet-600",
                    "disabled:cursor-not-allowed",
                    "disabled:opacity-60",
                    selected
                      ? "border-violet-600"
                      : "border-slate-200 hover:border-violet-300",
                  ].join(" ")}
                >
                  {image.previewUrl ? (
                    <img
                      src={image.previewUrl}
                      alt="Generated workspace image"
                      loading="lazy"
                      referrerPolicy="no-referrer"
                      className="h-full w-full object-cover"
                      onError={() => {
                        setImages((current) =>
                          current.map((item) =>
                            item.assetId === image.assetId
                              ? {
                                  ...item,
                                  previewUrl: null,
                                }
                              : item,
                          ),
                        );
                      }}
                    />
                  ) : (
                    <div className="flex h-full flex-col items-center justify-center gap-2 p-2 text-center">
                      <ImageIcon size={22} className="text-slate-400" />

                      <span className="text-xs text-slate-500">
                        Preview unavailable
                      </span>
                    </div>
                  )}

                  {selected && (
                    <span className="absolute right-2 top-2 rounded-full bg-white p-1 text-violet-700 shadow">
                      <CheckCircle2 size={19} />
                    </span>
                  )}
                </button>

                {scope === "all" && (
                  <p
                    className="truncate text-xs font-normal text-slate-600"
                    title={image.sourceWorkspaceName}
                  >
                    {image.sourceWorkspaceName}
                  </p>
                )}
              </div>
            );
          })}
        </div>
      )}

      <p className="text-xs leading-5 text-slate-500">
        Showing up to 12 recent ready images. Each image remains in its original
        workspace. Selecting an image does not consume AI credits.
      </p>
    </section>
  );
}

export function selectPostImage(

  return {

    selectedMediaIds: [assetId],
  };
}

export function removePostImage(state: PostMediaState): PostMediaState {
  return {
    ...state,

  };
}

export function hasPostMediaChanged(state: PostMediaState): boolean {
  if (state.originalMediaIds.length !== state.selectedMediaIds.length) {
    return true;
  }

  return state.originalMediaIds.some(
    (id, index) => id !== state.selectedMediaIds[index],
  );
}

export function getPostMediaUpdate(state: PostMediaState): {
  mediaIds?: string[];
} {
  if (!hasPostMediaChanged(state)) {
    return {};
  }

  return {
    mediaIds: [...state.selectedMediaIds],
  };
}

import { getImageStorageKey } from "./supabaseImageStorage.js";

export type ImageEligibilityInput = {
  _id: unknown;
  workspaceId: unknown;

  type?: unknown;
  source?: unknown;
  status?: unknown;

  storageProvider?: unknown;
  storageKey?: unknown;
  mimeType?: unknown;

  width?: unknown;
  height?: unknown;
  fileSize?: unknown;

  idempotencyKey?: unknown;
  generationExecution?: unknown;
  contentSha256?: unknown;
};

export type ImageEligibilityResult =
  | {
      eligible: true;
    }
  | {
      eligible: false;
      reason:
        | "invalid-identity"
        | "invalid-status"
        | "invalid-storage"
        | "invalid-file-metadata"
        | "missing-generation-verification";
    };

const OBJECT_ID_PATTERN = /^[a-f0-9]{24}$/i;
const SHA256_PATTERN = /^[a-f0-9]{64}$/;

function isPositiveInteger(value: unknown): boolean {
  return typeof value === "number" && Number.isSafeInteger(value) && value > 0;
}

export function evaluateImageEligibility(
  asset: ImageEligibilityInput,
): ImageEligibilityResult {
  const assetId = String(asset._id ?? "");
  const workspaceId = String(asset.workspaceId ?? "");

  if (
    !OBJECT_ID_PATTERN.test(assetId) ||
    !OBJECT_ID_PATTERN.test(workspaceId)
  ) {
    return {
      eligible: false,
      reason: "invalid-identity",
    };
  }

  if (
    asset.type !== "image" ||
    asset.source !== "ai-generated" ||
    asset.status !== "ready"
  ) {
    return {
      eligible: false,
      reason: "invalid-status",
    };
  }

  const expectedStorageKey = getImageStorageKey(workspaceId, assetId);

  if (
    asset.storageProvider !== "supabase" ||
    asset.storageKey !== expectedStorageKey ||
    asset.mimeType !== "image/png"
  ) {
    return {
      eligible: false,
      reason: "invalid-storage",
    };
  }

  if (
    !isPositiveInteger(asset.width) ||
    !isPositiveInteger(asset.height) ||
    !isPositiveInteger(asset.fileSize)
  ) {
    return {
      eligible: false,
      reason: "invalid-file-metadata",
    };
  }

  if (
    typeof asset.idempotencyKey !== "string" ||
    asset.idempotencyKey.trim().length === 0 ||
    !asset.generationExecution ||
    typeof asset.generationExecution !== "object" ||
    typeof asset.contentSha256 !== "string" ||
    !SHA256_PATTERN.test(asset.contentSha256)
  ) {
    return {
      eligible: false,
      reason: "missing-generation-verification",
    };
  }

  return {
    eligible: true,
  };
}

export function isPreviewReadyImage(asset: ImageEligibilityInput): boolean {
  return evaluateImageEligibility(asset).eligible;
}
