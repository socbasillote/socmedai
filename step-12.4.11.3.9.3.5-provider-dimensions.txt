import { createHash } from "node:crypto";
import sharp from "sharp";

import { generateImageSchema } from "../../../schemas/aiImageSchema.js";

import type {
  ImageProvider,
  ImageProviderRequest,
  ImageProviderResult,
} from "./imageProvider.js";

const DIMENSIONS = {
  square: {
    width: 1024,
    height: 1024,
  },
  portrait: {
    width: 1024,
    height: 1536,
  },
  landscape: {
    width: 1536,
    height: 1024,
  },
} as const;

/**
 * Generates genuine PNG bytes entirely locally.
 *
 * The same normalized request always produces
 * the same image, supporting deterministic tests.
 */
export class MockImageProvider implements ImageProvider {
  readonly provider = "mock";
  readonly model = "storm-mock-image-v1";

  async generate(request: ImageProviderRequest): Promise<ImageProviderResult> {
    const input = generateImageSchema.parse(request);

    const { width, height } = DIMENSIONS[input.aspectRatio];

    const hash = createHash("sha256")
      .update(JSON.stringify([input.prompt, input.aspectRatio]))
      .digest("hex");

    const background = `#${hash.slice(0, 6)}`;

    const bytes = await sharp({
      create: {
        width,
        height,
        channels: 3,
        background,
      },
    })
      .png()
      .toBuffer();

    return {
      bytes,
      mimeType: "image/png",
      provider: this.provider,
      model: this.model,
    };
  }
}

===== apps/api/src/services/media/providers/openaiImageProvider.ts =====
import { generateImageSchema } from "../../../schemas/aiImageSchema.js";

import {
  MAX_IMAGE_BYTES,
  MAX_IMAGE_DIMENSION,
  validateGeneratedImage,
} from "../imageValidation.js";

import {
  ImageProviderError,
  type ImageProvider,
  type ImageProviderRequest,
  type ImageProviderResult,
} from "./imageProvider.js";

const IMAGE_ENDPOINT = "https://api.openai.com/v1/images/generations";

const REQUEST_TIMEOUT_MS = 120_000;
const MAX_RESPONSE_BYTES = 30 * 1024 * 1024;

const IMAGE_SIZES = {
  square: "1024x1024",
  portrait: "1024x1536",
  landscape: "1536x1024",
} as const;

type OpenAIImageProviderOptions = {
  apiKey: string;
  model: string;

  /**
   * Required injection:
   * Tests provide a mock function.
   * The trusted production factory will
   * eventually provide global fetch.
   */
  fetchImpl: typeof fetch;
};

function providerError(
  message: string,
  outcome: "rejected" | "uncertain",
): ImageProviderError {
  return new ImageProviderError(message, outcome);
}

/**
 * Log only safe diagnostic metadata.
 *
 * Never log:
 * - API keys
 * - prompts
 * - response bodies
 * - generated images
 * - authorization headers
 */
function logProviderFailure(
  stage: string,
  details: {
    httpStatus?: number;
    errorType?: string;
  } = {},
): void {
  console.error("[OpenAI Image Provider]", {
    stage,
    ...details,
  });
}

/**
 * Read a bounded response body.
 *
 * Rejecting oversized responses prevents a
 * malformed upstream response from consuming
 * unlimited API-server memory.
 */
async function readBoundedResponse(response: Response): Promise<string> {
  if (!response.body) {
    return "";
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder();

  let total = 0;
  let output = "";

  try {
    while (true) {
      const { done, value } = await reader.read();

      if (done) break;

      total += value.byteLength;

      if (total > MAX_RESPONSE_BYTES) {
        throw providerError(
          "Image provider response exceeded the allowed size.",
          "uncertain",
        );
      }

      output += decoder.decode(value, {
        stream: true,
      });
    }

    output += decoder.decode();
    return output;
  } finally {
    reader.releaseLock();
  }
}

function classifyHttpError(status: number): ImageProviderError {
  return providerError(
    `Image provider request was not completed successfully (HTTP ${status}).`,
    "uncertain",
  );
}

function isStrictBase64(value: string): boolean {
  if (
    !value ||
    value.length % 4 !== 0 ||
    !/^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/.test(
      value,
    )
  ) {
    return false;
  }

  return true;
}

export class OpenAIImageProvider implements ImageProvider {
  readonly provider = "openai";
  readonly model: string;

  private readonly apiKey: string;
  private readonly fetchImpl: typeof fetch;

  constructor(options: OpenAIImageProviderOptions) {
    const apiKey = options.apiKey.trim();
    const model = options.model.trim();

    if (!apiKey) {
      throw new Error("OpenAI image provider credentials are unavailable.");
    }

    if (!model || model.length > 120 || !/^[A-Za-z0-9._-]+$/.test(model)) {
      throw new Error("Invalid OpenAI image model configuration.");
    }

    if (
      !/^gpt-image-(?:1-mini|1\.5|2(?:-[A-Za-z0-9.-]+)?|2\.5-(?:sunburst|flare)(?:-[A-Za-z0-9.-]+)?|2\.6-(?:sunburst|flare)(?:-[A-Za-z0-9.-]+)?)$/.test(
        model,
      )
    ) {
      throw new Error("Unsupported OpenAI image model configuration.");
    }

    if (typeof options.fetchImpl !== "function") {
      throw new Error("An HTTP transport is required.");
    }

    this.apiKey = apiKey;
    this.model = model;
    this.fetchImpl = options.fetchImpl;
  }

  async generate(request: ImageProviderRequest): Promise<ImageProviderResult> {
    const parsed = generateImageSchema.safeParse(request);

    if (!parsed.success) {
      throw providerError("Invalid image generation request.", "rejected");
    }

    const expectedSize = IMAGE_SIZES[parsed.data.aspectRatio];

    let response: Response;

    try {
      response = await this.fetchImpl(IMAGE_ENDPOINT, {
        method: "POST",
        headers: {
          Authorization: `Bearer ${this.apiKey}`,
          "Content-Type": "application/json",
          Accept: "application/json",
        },
        body: JSON.stringify({
          model: this.model,
          prompt: parsed.data.prompt,
          size: expectedSize,
          quality: "low",
          n: 1,
          output_format: "png",
        }),
        redirect: "error",
        signal: AbortSignal.timeout(REQUEST_TIMEOUT_MS),
      });
    } catch (error: unknown) {
      const errorType = error instanceof Error ? error.name : "UnknownError";

      logProviderFailure("transport", {
        errorType,
      });

      throw providerError(
        "Image provider request outcome is uncertain.",
        "uncertain",
      );
    }

    if (!response.ok) {
      logProviderFailure("http_response", {
        httpStatus: response.status,
      });

      await response.body?.cancel().catch(() => undefined);

      throw classifyHttpError(response.status);
    }

    let raw: string;

    try {
      raw = await readBoundedResponse(response);
    } catch {
      logProviderFailure("response_read");

      throw providerError(
        "Image provider response could not be read safely.",
        "uncertain",
      );
    }

    let payload: unknown;

    try {
      payload = JSON.parse(raw);
    } catch {
      logProviderFailure("invalid_json");

      throw providerError("Image provider returned invalid JSON.", "uncertain");
    }

    if (
      !payload ||
      typeof payload !== "object" ||
      !("data" in payload) ||
      !Array.isArray(payload.data) ||
      payload.data.length !== 1
    ) {
      throw providerError(
        "Image provider returned an unexpected image count.",
        "uncertain",
      );
    }

    const item: unknown = payload.data[0];

    if (
      !item ||
      typeof item !== "object" ||
      !("b64_json" in item) ||
      typeof item.b64_json !== "string"
    ) {
      throw providerError(
        "Image provider returned no image data.",
        "uncertain",
      );
    }

    const base64 = item.b64_json;

    if (
      base64.length > Math.ceil((MAX_IMAGE_BYTES * 4) / 3) + 4 ||
      !isStrictBase64(base64)
    ) {
      throw providerError(
        "Image provider returned invalid image encoding.",
        "uncertain",
      );
    }

    const decoded = Buffer.from(base64, "base64");

    if (decoded.length === 0 || decoded.length > MAX_IMAGE_BYTES) {
      throw providerError(
        "Image provider returned an oversized image.",
        "uncertain",
      );
    }

    try {
      const validated = await validateGeneratedImage(decoded);

      const [expectedWidth, expectedHeight] = expectedSize
        .split("x")
        .map(Number);

      if (
        validated.width !== expectedWidth ||
        validated.height !== expectedHeight ||
        validated.width > MAX_IMAGE_DIMENSION ||
        validated.height > MAX_IMAGE_DIMENSION
      ) {
        throw new Error("Unexpected generated image dimensions.");
      }

      return {
        bytes: validated.bytes,
        mimeType: "image/png",
        provider: this.provider,
        model: this.model,
      };
    } catch {
      logProviderFailure("image_validation");

      throw providerError(
        "Image provider returned an invalid or unexpected image.",
        "uncertain",
      );
    }
  }
}

===== apps/api/src/services/media/imageValidation.ts =====
import sharp from "sharp";
import { createHash } from "node:crypto";

export const MAX_IMAGE_BYTES = 20 * 1024 * 1024;
export const MAX_IMAGE_DIMENSION = 16_384;
export const MAX_IMAGE_PIXELS = 40_000_000;

export class ImageValidationError extends Error {
  constructor(message: string) {
    super(message);
    this.name = "ImageValidationError";
  }
}

export type ValidatedImage = {
  bytes: Buffer;
  mimeType: "image/png";
  width: number;
  height: number;
  fileSize: number;
  sha256: string;
};

export async function validateGeneratedImage(
  input: Uint8Array,
): Promise<ValidatedImage> {
  if (
    !(input instanceof Uint8Array) ||
    input.byteLength === 0 ||
    input.byteLength > MAX_IMAGE_BYTES
  ) {
    throw new ImageValidationError(
      "Image data is empty or exceeds the 20 MB limit.",
    );
  }

  try {
    const source = sharp(input, {
      failOn: "error",
      limitInputPixels: MAX_IMAGE_PIXELS,
      animated: false,
    });

    const metadata = await source.metadata();

    if (!["png", "jpeg", "webp"].includes(metadata.format ?? "")) {
      throw new ImageValidationError("Unsupported image format.");
    }

    if (
      !metadata.width ||
      !metadata.height ||
      metadata.width > MAX_IMAGE_DIMENSION ||
      metadata.height > MAX_IMAGE_DIMENSION ||
      metadata.width * metadata.height > MAX_IMAGE_PIXELS ||
      (metadata.pages ?? 1) !== 1
    ) {
      throw new ImageValidationError(
        "Invalid image dimensions or multiple frames.",
      );
    }

    // Re-encode to normalize the format and remove
    // source metadata before storing.
    const normalized = await sharp(input, {
      failOn: "error",
      limitInputPixels: MAX_IMAGE_PIXELS,
      animated: false,
    })
      .rotate()
      .png()
      .toBuffer({ resolveWithObject: true });

    const bytes = normalized.data;
    const { width, height } = normalized.info;

    if (
      width < 1 ||
      height < 1 ||
      width > MAX_IMAGE_DIMENSION ||
      height > MAX_IMAGE_DIMENSION ||
      bytes.length > MAX_IMAGE_BYTES
    ) {
      throw new ImageValidationError(
        "Normalized image exceeds storage limits.",
      );
    }

    return {
      bytes,
      mimeType: "image/png",
      width,
      height,
      fileSize: bytes.length,
      sha256: createHash("sha256").update(bytes).digest("hex"),
    };
  } catch (error) {
    if (error instanceof ImageValidationError) {
      throw error;
    }

    throw new ImageValidationError("Image data could not be decoded.");
  }
}

===== apps/api/src/services/media/textLabelLayout.ts =====
export type TextLabelLayoutInput = {
  text: string;
  imageWidth: number;
  imageHeight: number;
  sizePercent: number;
};

export type TextLabelLayout = {
  lines: string[];
  fontSize: number;
  overlayWidth: number;
  overlayHeight: number;
};

const MIN_FONT_SIZE = 12;
const MAX_LINES = 5;
const LINE_HEIGHT_MULTIPLIER = 1.5;

/**
 * Conservative character-width estimate.
 *
 * The final Sharp renderer will need actual
 * font-metric verification before release.
 */
const CHAR_WIDTH_FACTOR = 0.75;

export function planTextLabelLayout(
  input: TextLabelLayoutInput,
): TextLabelLayout {
  const { imageWidth, imageHeight, sizePercent } = input;

  const text = input.text.trim();

  if (!text || text.length > 80) {
    throw new Error("Invalid text-label content.");
  }

  if (
    !Number.isInteger(imageWidth) ||
    !Number.isInteger(imageHeight) ||
    imageWidth < 1 ||
    imageHeight < 1 ||
    !Number.isInteger(sizePercent) ||
    sizePercent < 5 ||
    sizePercent > 30
  ) {
    throw new Error("Invalid text-label dimensions.");
  }

  const margin = Math.max(
    4,
    Math.round(Math.min(imageWidth, imageHeight) * 0.025),
  );

  const overlayWidth = Math.min(
    imageWidth - margin * 2,
    Math.round((imageWidth * sizePercent) / 100),
  );

  const availableHeight = imageHeight - margin * 2;

  const padding = Math.max(2, Math.round(overlayWidth * 0.025));

  const availableWidth = overlayWidth - padding * 2;

  const maxCharactersPerLine = Math.floor(
    availableWidth / (MIN_FONT_SIZE * CHAR_WIDTH_FACTOR),
  );

  if (maxCharactersPerLine < 1) {
    throw new Error("Text label is too small to display readably.");
  }

  // This is an initial code-point-based layout
  // estimate, not a measured font layout.
  const characters = Array.from(text);

  const lines: string[] = [];

  for (
    let index = 0;
    index < characters.length;
    index += maxCharactersPerLine
  ) {
    lines.push(characters.slice(index, index + maxCharactersPerLine).join(""));
  }

  if (lines.length > MAX_LINES) {
    throw new Error("Text label is too long for the selected size.");
  }

  const fontSize = MIN_FONT_SIZE;

  const overlayHeight = Math.ceil(
    lines.length * fontSize * LINE_HEIGHT_MULTIPLIER,
  );

  if (overlayHeight > availableHeight) {
    throw new Error("Text label does not fit within the image.");
  }

  return {
    lines,
    fontSize,
    overlayWidth,
    overlayHeight,
  };
}

===== apps/api/src/services/media/verifyAdmittedImageBrandingPreflight.ts =====
import {
  resolveAdmittedImageBranding,
  type ResolveAdmittedImageBrandingInput,
} from "./resolveAdmittedImageBranding.js";

import { downloadVerifiedWorkspaceLogo } from "./workspaceLogoStorage.js";

import type { ImageStorageBucket } from "./supabaseImageStorage.js";

/**
 * Validate the admitted branding before
 * crossing the provider_started boundary.
 *
 * This service does not:
 * - call an AI provider
 * - debit credits
 * - update MongoDB
 * - upload generated images
 */
export async function verifyAdmittedImageBrandingPreflight(
  input: ResolveAdmittedImageBrandingInput,
  bucket: ImageStorageBucket,
): Promise<void> {
  const branding = resolveAdmittedImageBranding(input);

  if (branding.type === "none") {
    return;
  }

  if (branding.type === "text") {
    // Text is already validated by the resolver.
    //
    // Its renderer integration is not yet enabled
    // for admission. This branch prepares the
    // worker for that later checkpoint.
    return;
  }

  // Logo mode: verify the exact immutable
  // private object captured during admission.
  await downloadVerifiedWorkspaceLogo(
    {
      workspaceId: input.workspaceId,
      storageKey: branding.logo.storageKey,
      sha256: branding.logo.sha256,
    },
    bucket,
  );
}

import { z } from "zod";

/**
 * User-facing aspect ratios.
 *
 * The provider adapter translates these
 * into supported provider sizes.
 */
export const imageAspectRatioSchema = z.enum([
  "square",
  "portrait",
  "landscape",
]);

/**
 * Shared positioning options for logo
 * overlays and text labels.
 */
export const imageBrandingPositionSchema = z.enum([
  "top-left",
  "top-right",
  "bottom-left",
  "bottom-right",
]);

const brandingAppearanceSchema = {
  position: imageBrandingPositionSchema.default("bottom-right"),

  sizePercent: z.number().int().min(5).max(30).default(20),

  opacity: z.number().min(0.1).max(1).default(0.8),
};

/**
 * Branding is explicitly selected for each
 * individual image generation request.
 *
 * No branding is the default.
 */
export const imageBrandingSchema = z.discriminatedUnion("type", [
  z
    .object({
      type: z.literal("none"),
    })
    .strict(),

  z
    .object({
      type: z.literal("logo"),
      ...brandingAppearanceSchema,
    })
    .strict(),

  z
    .object({
      type: z.literal("text"),

      text: z
        .string()
        .trim()
        .min(1, "Enter a text label.")
        .max(80, "The label cannot exceed 80 characters.")
        .refine(
          (value) => !/[\u0000-\u001f\u007f]/u.test(value),
          "Text labels cannot contain control characters.",
        ),

      ...brandingAppearanceSchema,
    })
    .strict(),
]);

export const generateImageSchema = z
  .object({
    prompt: z
      .string()
      .trim()
      .min(10, "Describe the image in at least 10 characters.")
      .max(3000, "The image prompt cannot exceed 3,000 characters."),

    aspectRatio: imageAspectRatioSchema.default("square"),

    branding: imageBrandingSchema.default({ type: "none" }),
  })
  .strict();

export type ImageAspectRatio = z.infer<typeof imageAspectRatioSchema>;

export type ImageBranding = z.infer<typeof imageBrandingSchema>;

/**
 * Incoming image-generation request.
 *
 * Branding and aspectRatio can be omitted.
 * Zod supplies defaults during parsing.
 */
export type GenerateImageInput = z.input<typeof generateImageSchema>;

/**
 * Validated, normalized image-generation request.
 *
 * Defaults have already been applied.
 */
export type ParsedGenerateImageInput = z.output<typeof generateImageSchema>;
