function fakeBucket() {
  const objects = new Map<string, Uint8Array>();

  const upload = vi.fn(
    async (
      key: string,
      body: ArrayBuffer,
    ): Promise<{ error: { message: string } | null }> => {
      if (objects.has(key)) {
        return {
          error: { message: "Already exists" },
        };
      }

      objects.set(key, new Uint8Array(body));
      return { error: null };
    },
  );

  const download = vi.fn(
    async (
      key: string,
    ): Promise<{
      data: Blob | null;
      error: { message: string } | null;
    }> => {
      const bytes = objects.get(key);

      if (!bytes) {
        return {
          data: null,
          error: { message: "Not found" },
        };
      }

      return {
        data: new Blob([Uint8Array.from(bytes).buffer as ArrayBuffer]),
        error: null,
      };
    },
  );

  const remove = vi.fn(
    async (keys: string[]): Promise<{ error: { message: string } | null }> => {
      for (const key of keys) {
        objects.delete(key);
      }

      return { error: null };
    },
  );

  const bucket: ImageStorageBucket = {
    upload,
    download,
    remove,
  };

  return {
    bucket,
    objects,
    upload,
    download,
    remove,
  };
}

describe("Generation crash-safety hardening", () => {
  it("persists the checksum before upload begins", async () => {
    const context = await createStartedGeneration();
    const storage = fakeBucket();

    let observedSha256: string | undefined;
    let observedPhase: string | undefined;

    const originalUpload = storage.bucket.upload;

    storage.bucket.upload = async (key, body, options) => {
      const checkpoint = await MediaAsset.findById(context.assetId);

      observedSha256 = checkpoint?.contentSha256;

      observedPhase = checkpoint?.generationExecution?.phase;

      return originalUpload(key, body, options);
    };

    const completed = await completeClaimedImageGeneration(
      {
        ...context,
        bytes: await sampleImage(),
      },
      storage.bucket,
    );

    expect(observedPhase).toBe("storage_pending");
    expect(observedSha256).toMatch(/^[a-f0-9]{64}$/);

    expect(completed.contentSha256).toBe(observedSha256);
  });

  it("retains the checksum after an upload failure", async () => {
    const context = await createStartedGeneration();
    const storage = fakeBucket();

    storage.upload.mockImplementationOnce(async () => ({
      error: {
        message: "Simulated storage failure",
      },
    }));

    await expect(
      completeClaimedImageGeneration(
        {
          ...context,
          bytes: await sampleImage(),
        },
        storage.bucket,
      ),
    ).rejects.toThrow();

    const saved = await MediaAsset.findById(context.assetId);

    expect(saved?.status).toBe("pending");
    expect(saved?.generationExecution?.phase).toBe("storage_pending");

    expect(saved?.contentSha256).toMatch(/^[a-f0-9]{64}$/);
  });

  it("verifies a stored image after a simulated crash", async () => {
    const context = await createStartedGeneration();
    const storage = fakeBucket();

    const bytes = await sampleImage();

    // Simulate a crash after the durable checkpoint
    // and before normal MongoDB finalization.
    //
    // The mock records the uploaded object,
    // then throws to interrupt the workflow.
    const originalUpload = storage.bucket.upload;

    storage.bucket.upload = async (key, body, options) => {
      await originalUpload(key, body, options);

      throw new Error("Simulated process interruption");
    };

    await expect(
      completeClaimedImageGeneration(
        {
          ...context,
          bytes,
        },
        storage.bucket,
      ),
    ).rejects.toThrow();

    const beforeRecovery = await MediaAsset.findById(context.assetId);

    expect(beforeRecovery?.status).toBe("pending");

    const expectedSha = beforeRecovery?.contentSha256;

    expect(expectedSha).toMatch(/^[a-f0-9]{64}$/);

    // Simulate the original worker disappearing
    // and its lease subsequently expiring.
    await MediaAsset.collection.updateOne(
      {
        _id: new mongoose.Types.ObjectId(context.assetId),
      },
      {
        $set: {
          "generationExecution.leaseExpiresAt": new Date(Date.now() - 60_000),
        },
      },
    );

    const recovery = await inspectStoredGeneration(context, storage.bucket);

    expect(recovery.outcome).toBe("checksum_verified");

    expect(recovery.objectSha256).toBe(expectedSha);

    const afterRecovery = await MediaAsset.findById(context.assetId);

    // Inspection must not automatically
    // finalize or repeat paid generation.
    expect(afterRecovery?.status).toBe("pending");
    expect(afterRecovery?.generationExecution?.phase).toBe("recovery_required");
  });

  it("rejects a lease renewal with the wrong token", async () => {
    const context = await createStartedGeneration();

    await expect(
      renewGenerationLease({
        ...context,
        claimToken: "00000000-0000-4000-8000-000000000002",
      }),
    ).rejects.toMatchObject({
      statusCode: 409,
    });
  });

  it("renews an active owned lease", async () => {
    const context = await createStartedGeneration();

    const before = await MediaAsset.findById(context.assetId);

    const renewed = await renewGenerationLease(context);

    expect(renewed.generationExecution?.leaseExpiresAt).toBeInstanceOf(Date);

    expect(
      renewed.generationExecution!.leaseExpiresAt!.getTime(),
    ).toBeGreaterThanOrEqual(
      before!.generationExecution!.leaseExpiresAt!.getTime(),
    );
  });

  it("refuses to revive an expired lease", async () => {
    const context = await createStartedGeneration();

    await MediaAsset.collection.updateOne(
      {
        _id: new mongoose.Types.ObjectId(context.assetId),
      },
      {
        $set: {
          "generationExecution.leaseExpiresAt": new Date(Date.now() - 60_000),
        },
      },
    );

    await expect(renewGenerationLease(context)).rejects.toMatchObject({
      statusCode: 409,
    });
  });

  it("refuses lease renewal after recovery takes ownership", async () => {
    const context = await createStartedGeneration();

    await MediaAsset.collection.updateOne(
      {
        _id: new mongoose.Types.ObjectId(context.assetId),
      },
      {
        $set: {
          "generationExecution.phase": "recovery_required",
        },
      },
    );

    await expect(renewGenerationLease(context)).rejects.toMatchObject({
      statusCode: 409,
    });
  });

  it("checks ownership before and after a heartbeat operation", async () => {
    const context = await createStartedGeneration();

    let called = false;

    const value = await withGenerationLeaseHeartbeat(context, async () => {
      called = true;
      return "upload-completed";
    });

    expect(called).toBe(true);
    expect(value).toBe("upload-completed");

    const asset = await MediaAsset.findById(context.assetId);

    expect(asset?.status).toBe("pending");
    expect(asset?.generationExecution?.phase).toBe("provider_started");
  });

  it("rejects finalization when an object differs from its checkpoint", async () => {
    const context = await createStartedGeneration();
    const storage = fakeBucket();

    const originalImage = await sampleImage();

    const normalized = await validateGeneratedImage(originalImage);

    const originalUpload = storage.bucket.upload;

    storage.bucket.upload = async (key, body, options) => {
      // The normal upload adapter must verify
      // the stored bytes rather than trusting
      // the response from this mock.
      const result = await originalUpload(key, body, options);

      const differentImage = await sharp({
        create: {
          width: 128,
          height: 96,
          channels: 3,
          background: "#ef4444",
        },
      })
        .png()
        .toBuffer();

      storage.objects.set(key, differentImage);

      return result;
    };

    await expect(
      completeClaimedImageGeneration(
        {
          ...context,
          bytes: originalImage,
        },
        storage.bucket,
      ),
    ).rejects.toThrow();

    const saved = await MediaAsset.findById(context.assetId);

    expect(saved?.status).toBe("pending");
    expect(saved?.contentSha256).toBe(normalized.sha256);

    // The storage adapter attempts cleanup when
    // read-back integrity verification fails.
    expect(storage.remove).toHaveBeenCalled();

    expect(
      storage.objects.has(getImageStorageKey(workspaceA, context.assetId)),
    ).toBe(false);
  });
});

function fakeStorage() {
  const objects = new Map<string, Uint8Array>();

  const upload = vi.fn(

      data: Blob | null;

          data: null,
          error: { message: "Not found" },
        };
      }

      return {
        data: new Blob([Uint8Array.from(bytes).buffer as ArrayBuffer]),
        error: null,

  const remove = vi.fn(
    async (

    async (

      return {
        data: new Blob([Uint8Array.from(bytes).buffer as ArrayBuffer]),
        error: null,
      };
    },
  );

  const remove = vi.fn(
    async (
      paths: string[],
    ): Promise<{
      error: { message: string } | null;
    }> => {
      for (const path of paths) {
        objects.delete(path);
      }

      return { error: null };
    },
  );

  return {
    bucket: {
      upload,
      download,
      remove,
    } satisfies ImageStorageBucket,
    objects,
    upload,
    download,
    remove,
  };

  const capacity = await ImageWorkspaceCapacity.findOne({
    workspaceId,
  });

  return capacity?.active ?? 0;
}

function context(assetId: string) {
  return {
    authUser: user,
    workspaceId,
    assetId,
  };
}

/**
 * Simulate a crash after the image has been
 * written, but before MongoDB finalization.
 */
async function createInterruptedUpload() {
  const queued = await enqueue();
  const s = storage();

  s.upload.mockImplementationOnce(async (path: string, bytes: ArrayBuffer) => {
    // Simulate an upload that successfully
    // writes the object to storage.
    s.objects.set(path, Uint8Array.from(new Uint8Array(bytes)));

    // But the client never receives confirmation.
    // This represents an ambiguous upload result.
    throw new Error("Simulated lost upload response");
  });

  const result = await runImageGenerationWorkerOnce({
    provider,
    bucket: s.bucket,
  });

  expect(result).toContainEqual({
    outcome: "recovery_required",
    assetId: queued.assetId,
  });

  const before = await MediaAsset.findById(queued.assetId);

  expect(before?.status).toBe("pending");

  expect(before?.generationExecution?.phase).toBe("storage_pending");

  expect(before?.contentSha256).toMatch(/^[a-f0-9]{64}$/);

  await expireLease(queued.assetId);

  return {
    queued,
    storage: s,
  };
}

describe("Generation crash reconciliation", () => {
  it("reclaims an expired pre-provider claim", async () => {
    const queued = await enqueue();

    const firstClaim = await claimImageGeneration(context(queued.assetId));

    expect(firstClaim.claimToken).toBeTruthy();

    await expireLease(queued.assetId);

    const s = storage();

    const results = await runImageGenerationWorkerOnce({
      provider,
      bucket: s.bucket,
    });

    expect(results).toContainEqual({
      assetId: queued.assetId,
      outcome: "completed",
    });

    const final = await MediaAsset.findById(queued.assetId);

    expect(final?.status).toBe("ready");

    expect(final?.generationExecution?.attemptCount).toBe(2);

    expect(await balance()).toBe(4);
    expect(await activeCount()).toBe(0);
  });

  it("never rediscovers expired provider_started jobs", async () => {
    const queued = await enqueue();

    const claimed = await claimImageGeneration(context(queued.assetId));

    await beginImageProviderCall({
      ...context(queued.assetId),
      claimToken: claimed.claimToken,
    });

    await expireLease(queued.assetId);

    const s = storage();

    const results = await runImageGenerationWorkerOnce({
      provider,
      bucket: s.bucket,
    });

    expect(results).toEqual([{ outcome: "idle" }]);

    const final = await MediaAsset.findById(queued.assetId);

    expect(final?.status).toBe("pending");

    expect(final?.generationExecution?.phase).toBe("provider_started");

    expect(await activeCount()).toBe(1);
  });

  it("recovers uploaded bytes using the durable SHA-256", async () => {
    const interrupted = await createInterruptedUpload();

    const reconciled = await reconcileVerifiedImageGeneration(
      context(interrupted.queued.assetId),
      interrupted.storage.bucket,
    );

    expect(reconciled.outcome).toBe("verified_ready");

    const asset = await MediaAsset.findById(interrupted.queued.assetId);

    expect(asset?.status).toBe("ready");

    expect(asset?.storageKey).toBe(
      getImageStorageKey(workspaceId, interrupted.queued.assetId),
    );

    expect(await balance()).toBe(4);

    // Image finalization and capacity repair
    // are separate idempotent steps.
    expect(await activeCount()).toBe(1);
  });

  it("does not finalize an image with corrupted stored bytes", async () => {
    const interrupted = await createInterruptedUpload();

    const key = getImageStorageKey(workspaceId, interrupted.queued.assetId);

    interrupted.storage.objects.set(key, Buffer.from("corrupted image"));

    const reconciled = await reconcileVerifiedImageGeneration(
      context(interrupted.queued.assetId),
      interrupted.storage.bucket,
    );

    expect(reconciled.outcome).toBe("manual_review_required");

    const asset = await MediaAsset.findById(interrupted.queued.assetId);

    expect(asset?.status).toBe("pending");

    expect(await activeCount()).toBe(1);
  });

  it("cannot finalize a missing stored image", async () => {
    const interrupted = await createInterruptedUpload();

    interrupted.storage.objects.clear();

    const reconciled = await reconcileVerifiedImageGeneration(
      context(interrupted.queued.assetId),
      interrupted.storage.bucket,
    );

    expect(reconciled.outcome).toBe("manual_review_required");

    const asset = await MediaAsset.findById(interrupted.queued.assetId);

    expect(asset?.status).toBe("pending");
    expect(await balance()).toBe(4);
  });

  it("allows only one concurrent checksum recovery finalization", async () => {
    const interrupted = await createInterruptedUpload();

    const results = await Promise.all([
      reconcileVerifiedImageGeneration(
        context(interrupted.queued.assetId),
        interrupted.storage.bucket,
      ),
      reconcileVerifiedImageGeneration(
        context(interrupted.queued.assetId),
        interrupted.storage.bucket,
      ),
    ]);

    expect(results.some((r) => r.outcome === "verified_ready")).toBe(true);

    const asset = await MediaAsset.findById(interrupted.queued.assetId);

    expect(asset?.status).toBe("ready");

    expect(
      await MediaAsset.countDocuments({
        _id: interrupted.queued.assetId,
        status: "ready",
      }),
    ).toBe(1);

    expect(await balance()).toBe(4);
  });

  it("repairs capacity after terminal persistence", async () => {
    const interrupted = await createInterruptedUpload();

    await reconcileVerifiedImageGeneration(
      context(interrupted.queued.assetId),
      interrupted.storage.bucket,
    );

    const candidates = await discoverTerminalAdmissions();

    expect(candidates.map((c) => String(c.assetId))).toContain(
      interrupted.queued.assetId,
    );

    const repaired = await reconcileTerminalImageCapacity();

    expect(repaired).toContainEqual({
      assetId: interrupted.queued.assetId,
      outcome: "released",
    });

    expect(await activeCount()).toBe(0);
    expect(await balance()).toBe(4);

    const admission = await ImageGenerationAdmission.findOne({
      assetId: interrupted.queued.assetId,
    });

    expect(admission?.state).toBe("released");
  });

  it("does not release terminal capacity twice", async () => {
    const queued = await enqueue();

    await MediaAsset.updateOne(
      { _id: queued.assetId },
      {
        $set: {
          status: "failed",
        },
      },
    );

    const first = await reconcileTerminalImageCapacity();

    const second = await reconcileTerminalImageCapacity();

    expect(first).toContainEqual({
      assetId: queued.assetId,
      outcome: "released",
    });

    expect(second).toHaveLength(0);

    expect(await activeCount()).toBe(0);
    expect(await balance()).toBe(4);
  });

  it("handles concurrent terminal capacity repairs", async () => {
    const queued = await enqueue();

    await MediaAsset.updateOne(
      { _id: queued.assetId },
      {
        $set: {
          status: "failed",
        },
      },
    );

    await Promise.all([
      reconcileTerminalImageCapacity(),
      reconcileTerminalImageCapacity(),
    ]);

    expect(await activeCount()).toBe(0);
    expect(await balance()).toBe(4);

    const admission = await ImageGenerationAdmission.findOne({
      assetId: queued.assetId,
    });

    expect(admission?.state).toBe("released");
  });

  it("refuses recovery while a worker lease is active", async () => {
    const queued = await enqueue();

    const claimed = await claimImageGeneration(context(queued.assetId));

    await beginImageProviderCall({
      ...context(queued.assetId),
      claimToken: claimed.claimToken,
    });

    const reconciled = await reconcileVerifiedImageGeneration(
      context(queued.assetId),
      storage().bucket,
    );

    expect(reconciled.outcome).toBe("lease_active");

    const asset = await MediaAsset.findById(queued.assetId);

    expect(asset?.generationExecution?.phase).toBe("provider_started");

    expect(await activeCount()).toBe(1);
  });

  it("sweeps and finalizes a verified interrupted upload", async () => {
    const interrupted = await createInterruptedUpload();

    const sweep = await runImageGenerationRecoverySweep(
      interrupted.storage.bucket,
    );

    expect(sweep.inspected).toBe(1);
    expect(sweep.verifiedReady).toBe(1);
    expect(sweep.capacityReleased).toBe(1);

    const asset = await MediaAsset.findById(interrupted.queued.assetId);

    expect(asset?.status).toBe("ready");
    expect(await activeCount()).toBe(0);
    expect(await balance()).toBe(4);

    const second = await runImageGenerationRecoverySweep(
      interrupted.storage.bucket,
    );

    expect(second.inspected).toBe(0);
    expect(second.capacityReleased).toBe(0);
  });

  it("never regenerates an expired provider-started job during recovery", async () => {
    const queued = await enqueue();

    const claimed = await claimImageGeneration(context(queued.assetId));

    await beginImageProviderCall({
      ...context(queued.assetId),
      claimToken: claimed.claimToken,
    });

    await expireLease(queued.assetId);

    const s = storage();

    const sweep = await runImageGenerationRecoverySweep(s.bucket);

    expect(sweep.inspected).toBe(1);
    expect(sweep.verifiedReady).toBe(0);
    expect(sweep.needsReview).toBe(1);

    expect(s.upload).not.toHaveBeenCalled();

    const asset = await MediaAsset.findById(queued.assetId);

    expect(asset?.status).toBe("pending");

    expect(asset?.generationExecution?.phase).toBe("recovery_required");

    expect(await activeCount()).toBe(1);
    expect(await balance()).toBe(4);
  });
});
